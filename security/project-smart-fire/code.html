核心算法代码

算法说明：采用多传感器融合判据，对火焰、烟雾、温度三项指标综合打分， 根据得分等级执行不同响应策略。得分 ≥ 3 触发三级火警，自动启动灭火流程。 
#include "fire_alarm.h"

/* 传感器数据结构 */
typedef struct {
    uint8_t  flame_active;        // 火焰传感器触发
    uint16_t smoke_ppm;           // 烟雾浓度 (ppm)
    int16_t  temperature;         // 温度值 (°C * 10)
} SensorData;

/* 火情等级 */
typedef enum {
    FIRE_NONE  = 0,              // 正常
    FIRE_LEVEL1,                 // 注意：单传感器异常
    FIRE_LEVEL2,                 // 预警：双传感器异常
    FIRE_LEVEL3                  // 火警：三传感器异常 → 自动灭火
} FireLevel;

/* 多传感器融合判据 */
FireLevel Fire_Judge(SensorData *data) {
    uint8_t score = 0;

    // 火焰检测：任意 2 路以上触发计 1 分
    if (data->flame_active) score++;

    // 烟雾检测：浓度 > 300ppm 计 1 分
    if (data->smoke_ppm > 300) score++;

    // 温度检测：> 60°C 或 10s 温升 > 15°C 计 1 分
    if (data->temperature > 600) score++;

    // 综合判定
    switch (score) {
        case 0:  return FIRE_NONE;
        case 1:  return FIRE_LEVEL1;
        case 2:  return FIRE_LEVEL2;
        case 3:  return FIRE_LEVEL3;
        default: return FIRE_LEVEL3;
    }
}

/* 火情响应执行 */
void Fire_Response(FireLevel level) {
    switch (level) {
        case FIRE_NONE:
            Buzzer_Off();
            Relay_Off(RELAY_PUMP);
            break;

        case FIRE_LEVEL1:
            Buzzer_Beep(BEEP_INTERVAL_2S);
            LCD_ShowString("注意：检测到异常");
            break;

        case FIRE_LEVEL2:
            Buzzer_Beep(BEEP_INTERVAL_500MS);
            Relay_On(RELAY_EXHAUST);
            LCD_ShowString("预警：启动排烟");
            break;

        case FIRE_LEVEL3:
            Buzzer_Beep(BEEP_CONTINUOUS);
            Relay_On(RELAY_PUMP);              // 启动水泵喷淋
            Relay_On(RELAY_GAS_VALVE_OFF);     // 关闭燃气阀
            Relay_On(RELAY_MAIN_POWER_OFF);    // 切断电源
            GSM_SendAlert("FIRE_ALERT",
                "火警！请立即撤离！");         // 发送短信
            LCD_ShowString("火警！！自动灭火");
            break;
    }
}

/* 主循环（每 500ms 执行一次） */
void Fire_MainLoop(void) {
    SensorData sensor;

    sensor.flame_active = FlameSensor_ReadAny();
    sensor.smoke_ppm    = MQ2_ReadPPM();
    sensor.temperature  = DS18B20_ReadAvg();

    FireLevel level = Fire_Judge(&sensor);
    Fire_Response(level);

    printf("[FIRE] level=%d flame=%d smoke=%dppm temp=%d.%dC\n",
        level, sensor.flame_active, sensor.smoke_ppm,
        sensor.temperature / 10, sensor.temperature % 10);
}
