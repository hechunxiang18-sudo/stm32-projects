核心固件代码

认证逻辑：三种认证方式并行，任一通过即开门。IC 卡最快（0.3s），指纹次之（0.8s），密码需完整输入。
/* 门禁主控逻辑 */
void Door_Process(void) {
    /* 优先级：IC卡 > 指纹 > 密码 */
    if (RC522_CheckCard()) {
        if (RC522_Authenticate()) {
            Door_Open(OPEN_MODE_CARD);
            return;
        }
    }

    if (AS608_IsFingerPressed()) {
        uint16_t id; uint8_t score;
        if (AS608_Search(&id, &score) == AS608_OK && score > 60) {
            Door_Open(OPEN_MODE_FINGER);
            return;
        }
    }

    if (keyboard_input_ready) {
        if (Password_Verify(input_buf, input_len)) {
            Door_Open(OPEN_MODE_PASSWORD);
            return;
        }
    }

    /* 胁迫密码检测（特定码触发静默报警） */
    if (Password_IsDuress(input_buf, input_len)) {
        GSM_SendAlert("DURESS_ALERT", "胁迫开门！");
        Door_Open(OPEN_MODE_DURESS);
    }
}

void Door_Open(uint8_t mode) {
    HAL_GPIO_WritePin(DOOR_LOCK_PIN, GPIO_PIN_SET);
    Log_Entry(mode);              // 记录开门方式
    OLED_ShowIcon(mode);
    delay_ms(5000);               // 5 秒后自动闭锁
    HAL_GPIO_WritePin(DOOR_LOCK_PIN, GPIO_PIN_RESET);
    OLED_ShowStatus("已闭锁");
}
