核心固件代码

代码说明：以下展示指纹识别与密码验证的核心逻辑。 系统采用三种认证方式并行设计，按优先级依次尝试，任一通过即开锁。
lock.c — 指纹识别与密码验证核心逻辑
#include "lock.h"

/* 用户数据库（FLASH 存储） */
User_DB user_db[MAX_USERS];
uint32_t user_count = 0;

/* 开锁尝试计数（防暴力破解） */
static uint8_t fail_count = 0;
static uint32_t lock_start_time = 0;

/* 指纹匹配 */
Lock_Status Fingerprint_Verify(void) {
    if (!AS608_CheckFinger()) return LOCK_FAIL;
    uint16_t finger_id;
    uint8_t score;
    if (AS608_Search(&finger_id, &score) != AS608_OK)
        return LOCK_FAIL;
    if (score < 80) return LOCK_FAIL;      // 匹配度阈值
    return LOCK_OK;
}

/* 密码验证（支持虚位密码） */
Lock_Status Password_Verify(uint8_t *input, uint8_t len) {
    if (len < 6) return LOCK_FAIL;         // 最短 6 位

    // 提取最后 6 位为真实密码（虚位密码防偷窥）
    uint8_t *real_input = input + len - 6;

    uint8_t stored_pwd[6];
    FLASH_Read(FLASH_PWD_ADDR, stored_pwd, 6);

    for (int i = 0; i < 6; i++) {
        if (real_input[i] != stored_pwd[i])
            return LOCK_FAIL;
    }
    return LOCK_OK;
}

/* 综合开锁验证 */
Lock_Status Lock_Authenticate(void) {
    // 检查是否处于锁定状态
    if (lock_start_time > 0) {
        uint32_t elapsed = (HAL_GetTick() - lock_start_time) / 1000;
        if (elapsed < 30) return LOCK_BLOCKED;  // 锁定 30s
        else lock_start_time = 0;                 // 解锁
    }

    Lock_Status ret = LOCK_FAIL;

    // 按优先级尝试三种开锁方式
    if (Fingerprint_Available() && Fingerprint_Verify() == LOCK_OK)
        ret = LOCK_OK;

    if (ret != LOCK_OK && RFID_Available() && RFID_Verify() == LOCK_OK)
        ret = LOCK_OK;

    if (ret != LOCK_OK && Password_Available() &&
        Password_Verify(input_buf, input_len) == LOCK_OK)
        ret = LOCK_OK;

    if (ret == LOCK_OK) {
        fail_count = 0;
        Door_Unlock();             // 驱动电磁锁开锁
        Log_Entry(LOG_UNLOCK);     // 记录开锁日志
    } else {
        fail_count++;
        if (fail_count >= 5) {     // 5 次失败锁定 30 秒
            lock_start_time = HAL_GetTick();
            Buzzer_Alert(ALERT_LOCKED);
        }
    }
    return ret;
}

/* 驱动电磁锁 */
void Door_Unlock(void) {
    HAL_GPIO_WritePin(LOCK_GPIO_Port, LOCK_Pin, GPIO_PIN_SET);  // MOSFET 导通
    OLED_ShowString("✓ 已开锁");
    delay_ms(3000);                                              // 3 秒后自动闭锁
    HAL_GPIO_WritePin(LOCK_GPIO_Port, LOCK_Pin, GPIO_PIN_RESET);
    OLED_ShowString("已闭锁");
}
