核心代码 - NMEA 解析与 GSM 短信

代码说明：NMEA 协议解析 $GPGGA 语句提取经纬度， 通过 SIM800L AT+CMGS 指令发送短信位置。SD 卡 FATFS 文件系统存储 CSV 轨迹。 
/* NMEA 解析与 GSM 短信发送 (HAL 库) */

#include <stdio.h>
#include <string.h>
#include "fatfs.h"

/* GPS NMEA 缓冲区 */
#define NMEA_BUF  128
static char nmea_buf[NMEA_BUF];
static uint8_t nmea_idx = 0;

/* 解析后的坐标 */
typedef struct {
    float lat;     /* 纬度 */
    float lng;     /* 经度 */
    char  ns, ew;  /* N/S, E/W */
    uint8_t fix;   /* 定位状态 */
    uint8_t sat;   /* 卫星数 */
} GPS_t;

GPS_t gps;

/* NMEA 字符接收中断 */
void USART1_IRQHandler(void) {
    char c = USART1->DR;
    if (c == '\n') {
        nmea_buf[nmea_idx] = '\0';
        if (strstr(nmea_buf, "$GPGGA")) {
            Parse_GPGGA(nmea_buf, &gps);
        }
        nmea_idx = 0;
    } else {
        if (nmea_idx < NMEA_BUF - 1)
            nmea_buf[nmea_idx++] = c;
    }
}

/* 解析 $GPGGA 语句 */
void Parse_GPGGA(char *buf, GPS_t *gps) {
    char *p = buf;
    int field = 0;
    while (*p) {
        if (*p == ',') field++;
        else if (field == 2) {
            gps->lat = atof(p);    /* 纬度 ddmm.mmmm */
        } else if (field == 3) {
            gps->ns = *p;          /* N/S */
        } else if (field == 4) {
            gps->lng = atof(p);   /* 经度 dddmm.mmmm */
        } else if (field == 5) {
            gps->ew = *p;
        } else if (field == 6) {
            gps->fix = atoi(p);
        } else if (field == 7) {
            gps->sat = atoi(p);
            break;
        }
        p++;
    }
}

/* SIM800L 发送短信 */
void GSM_SendSMS(const char *phone, const char *msg) {
    char cmd[64];
    sprintf(cmd, "AT+CMGS=\"%s\"\r\n", phone);
    SIM800L_SendCmd(cmd);
    HAL_Delay(500);
    SIM800L_SendCmd(msg);
    HAL_Delay(200);
    SIM800L_SendByte(0x1A); /* Ctrl+Z 发送 */
    HAL_Delay(3000);
}

/* 生成位置短信并发送 */
void ReportPosition(GPS_t *gps) {
    char sms[128];
    snprintf(sms, sizeof(sms),
        "位置: %.4f%c, %.4f%c | 卫星: %d | %s",
        gps->lat, gps->ns, gps->lng, gps->ew,
        gps->sat, (gps->fix > 0) ? "已定位" : "未定位");
    GSM_SendSMS("13800138000", sms);
}

/* 电子围栏检查 */
void Fence_Check(GPS_t *gps, float lat_min, float lat_max,
                 float lng_min, float lng_max) {
    if (gps->lat < lat_min || gps->lat > lat_max ||
        gps->lng < lng_min || gps->lng > lng_max) {
        GSM_SendSMS("13800138000", "警报: 超出电子围栏!");
    }
}

/* 轨迹记录到 SD 卡 */
void Log_Track(GPS_t *gps) {
    FIL fil;
    char line[64];
    snprintf(line, sizeof(line), "%.4f,%c,%.4f,%c,%d\r\n",
        gps->lat, gps->ns, gps->lng, gps->ew, gps->sat);
    f_open(&fil, "track.csv", FA_OPEN_APPEND | FA_WRITE);
    f_write(&fil, line, strlen(line), NULL);
    f_close(&fil);
}
