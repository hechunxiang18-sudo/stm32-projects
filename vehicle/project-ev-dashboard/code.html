核心代码 - 速度测量与里程计算

代码说明：利用 TIM2 输入捕获功能测量霍尔传感器脉冲周期，结合车轮周长计算实时速度，积分计算里程。 
/* 电动车仪表 - 速度测量与里程计算 */

#include "speed_meter.h"
#include "eeprom_sim.h"

#define WHEEL_CIRCUMFERENCE_CM  188.4f   /* 60/70-12 轮胎周长 ≈ 1.884m */
#define PULSE_PER_REVOLUTION    1        /* 霍尔每圈一个脉冲 */
#define SPEED_FILTER_COEFF      0.15f    /* 低通滤波系数 */

static volatile uint32_t last_capture = 0;
static volatile uint32_t current_period = 0;
static float speed_kmh = 0.0f;
static float total_km = 0.0f;

/* TIM2 输入捕获中断 */
void TIM2_IRQHandler(void) {
    if (TIM2->SR & TIM_SR_CC1IF) {
        uint32_t capture = TIM2->CCR1;
        if (last_capture != 0) {
            current_period = capture - last_capture;
        }
        last_capture = capture;
    }
    TIM2->SR = 0;
}

/* 计算实时速度 (km/h) */
float speed_calc_current(void) {
    if (current_period == 0) return 0.0f;

    /* 频率 = 72MHz / period (Hz) */
    float freq = 72000000.0f / current_period;

    /* 线速度 = 频率 × 周长 (m/s) */
    float v_ms = freq * (WHEEL_CIRCUMFERENCE_CM / 100.0f);

    /* 转 km/h */
    float v_kmh = v_ms * 3.6f;

    /* 低通滤波 */
    speed_kmh = speed_kmh * (1 - SPEED_FILTER_COEFF)
              + v_kmh * SPEED_FILTER_COEFF;

    return speed_kmh;
}

/* 里程累加 (每 100ms 调用) */
void odometer_update(void) {
    static uint32_t tick_count = 0;
    tick_count++;

    /* 每秒累加一次 */
    if (tick_count >= 10) {
        tick_count = 0;
        float speed = speed_calc_current();
        /* 里程增量 = 速度 × 时间 (1s = 1/3600 h) */
        float delta_km = speed / 3600.0f;
        total_km += delta_km;

        /* 每 100m 保存一次到 EEPROM */
        static float last_save = 0;
        if (total_km - last_save >= 0.1f) {
            last_save = total_km;
            eeprom_save_odometer(total_km);
        }
    }
}

/* 初始化速度测量 */
void speed_meter_init(void) {
    /* TIM2 配置: 72MHz, 输入捕获模式 */
    RCC->APB1ENR |= RCC_APB1ENR_TIM2EN;
    TIM2->PSC = 0;                     /* 不分频, 72MHz 计数 */
    TIM2->ARR = 0xFFFFFFFF;            /* 最大自动重载 */
    TIM2->CCMR1 = TIM_CCMR1_CC1S_0;    /* 输入捕获 CC1 映射到 TI1 */
    TIM2->CCER = TIM_CCER_CC1E;        /* 捕获上升沿 */
    TIM2->DIER |= TIM_DIER_CC1IE;      /* 使能捕获中断 */
    NVIC_EnableIRQ(TIM2_IRQn);
    TIM2->CR1 |= TIM_CR1_CEN;          /* 启动定时器 */

    total_km = eeprom_load_odometer();
}
