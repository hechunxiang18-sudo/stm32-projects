核心代码 - CAN 通信与 OBD-II PID 解析

代码说明：通过 MCP2515 SPI 驱动收发 CAN 报文，遵循 ISO 15765-4 协议 发送 OBD-II PID 请求并解析车辆响应数据。 
/* MCP2515 CAN + OBD-II PID 读取 (HAL 库) */

#include "mcp2515.h"

/* OBD-II 常用 PID */
#define PID_ENGINE_RPM    0x0C
#define PID_VEHICLE_SPEED 0x0D
#define PID_COOLANT_TEMP  0x05
#define PID_FUEL_LEVEL    0x2F
#define PID_INTAKE_TEMP   0x0F

/* CAN 帧结构 */
typedef struct {
    uint32_t id;
    uint8_t  dlc;
    uint8_t  data[8];
} CAN_Frame_t;

/* 发送 OBD PID 请求 */
uint8_t OBD_RequestPID(uint8_t pid, uint8_t *result) {
    CAN_Frame_t tx, rx;

    tx.id  = 0x7DF;        /* OBD-II 广播地址 */
    tx.dlc = 8;
    tx.data[0] = 0x02;     /* 1 字节 PCI */
    tx.data[1] = 0x01;     /* 服务: 显示当前数据 */
    tx.data[2] = pid;
    tx.data[3] = 0xCC; tx.data[4] = 0xCC;
    tx.data[5] = 0xCC; tx.data[6] = 0xCC; tx.data[7] = 0xCC;

    MCP2515_SendFrame(&tx);

    /* 等待响应 (100ms 超时) */
    if (MCP2515_ReceiveFrame(&rx, 100)) {
        if ((rx.data[0] == 0x03 || rx.data[0] == 0x04)
            && rx.data[1] == 0x41
            && rx.data[2] == pid) {
            *result = rx.data[3];
            return 1;
        }
    }
    return 0;
}

/* 读取发动机转速 (RPM) */
uint16_t OBD_GetRPM(void) {
    uint8_t raw;
    if (OBD_RequestPID(PID_ENGINE_RPM, &raw)) {
        return raw * 4;  /* A*256+B / 4 */
    }
    return 0;
}

/* 读取车速 (km/h) */
uint8_t OBD_GetSpeed(void) {
    uint8_t raw;
    if (OBD_RequestPID(PID_VEHICLE_SPEED, &raw)) {
        return raw;
    }
    return 0;
}

/* 读取水温 (°C) */
int8_t OBD_GetCoolantTemp(void) {
    uint8_t raw;
    if (OBD_RequestPID(PID_COOLANT_TEMP, &raw)) {
        return raw - 40;  /* A-40 = °C */
    }
    return -40;
}

/* MCP2515 初始化 */
void MCP2515_Init_CAN(void) {
    MCP2515_Reset();
    MCP2515_SetMode(MCP2515_MODE_CONFIG);
    /* 500kbps, 16MHz 晶振 */
    MCP2515_SetBitrate(CAN_500KBPS, MCP_16MHZ);
    MCP2515_SetMode(MCP2515_MODE_NORMAL);
    MCP2515_SetFilter(0, 0x7E8, MCP2515_RXF0); /* 接收 ECU 响应 */
    MCP2515_SetMask(0, 0x7F8, MCP2515_RXM0);
}
