核心代码 - PID 温度控制与休眠管理

代码说明：ADC 采集 NTC 温度值，PID 计算 PWM 占空比控制 MOS 加热。MPU6050 检测静止/运动状态切换工作模式。 
/* 智能电烙铁 - PID 温度控制与休眠管理 */

#include "soldering_control.h"
#include "ntc_temp.h"
#include "pid.h"
#include "mpu6050.h"
#include "encoder.h"
#include "oled_display.h"

/* 工作状态 */
typedef enum {
    WORK_NORMAL,      /* 正常工作 */
    WORK_SLEEP,       /* 休眠待机 (200°C) */
    WORK_BOOST,       /* 快速回温中 */
    WORK_CALIBRATE    /* 校准模式 */
} WorkState;

static WorkState state = WORK_NORMAL;
static uint16_t set_temp = 350;   /* 设定温度 */
static uint16_t actual_temp = 0;
static uint32_t idle_seconds = 0;

/* PID 控制器实例 */
static PID_TypeDef pid;

/* NTC 温度读取 (经运放调理) */
uint16_t ntc_read_temperature(void) {
    uint32_t sum = 0;
    for (int i = 0; i < 16; i++) {
        ADC1->CR2 |= ADC_CR2_SWSTART;   /* 启动转换 */
        while (!(ADC1->SR & ADC_SR_EOC));
        sum += ADC1->DR;
    }
    uint16_t adc_val = sum / 16;

    /* NTC 分压 → 电阻 → 温度 (查表法) */
    float ntc_r = ntc_calc_resistance(adc_val);
    return ntc_lookup_temperature(ntc_r);
}

/* PID 初始化 */
void pid_configure(void) {
    PID_Init(&pid);
    pid.Kp = 2.8f;
    pid.Ki = 0.12f;
    pid.Kd = 0.35f;
    pid.IntegralLimit = 50.0f;
    pid.OutputLimit = 100.0f;
    PID_SetTarget(&pid, (float)set_temp);
}

/* 加热控制 */
void heater_set_power(uint8_t percent) {
    /* PWM 占空比控制 MOS */
    uint32_t pulse = (uint32_t)7199 * percent / 100;
    TIM2->CCR1 = pulse;
}

/* 主控循环 (周期 100ms) */
void soldering_main_loop(void) {
    actual_temp = ntc_read_temperature();

    switch (state) {
        case WORK_NORMAL:
            /* 读取编码器设定值 */
            int16_t enc_delta = encoder_get_delta();
            if (enc_delta != 0) {
                int16_t new_temp = set_temp + enc_delta * 5;
                if (new_temp < 100)  new_temp = 100;
                if (new_temp > 400)  new_temp = 400;
                set_temp = (uint16_t)new_temp;
                PID_SetTarget(&pid, (float)set_temp);
            }

            /* PID 计算并输出 */
            float pid_out = PID_Compute(&pid, (float)actual_temp);
            heater_set_power((uint8_t)pid_out);
            idle_seconds = 0;
            break;

        case WORK_SLEEP:
            /* 休眠保持 200°C */
            if (actual_temp < 200) {
                heater_set_power(20);   /* 低功率维持 */
            } else {
                heater_set_power(0);
            }
            break;

        case WORK_BOOST:
            /* 快速回温: 全功率加热 */
            heater_set_power(100);
            if (actual_temp >= set_temp - 5) {
                state = WORK_NORMAL;
            }
            break;
    }

    /* 更新 OLED */
    oled_display_temp(set_temp, actual_temp, state);
}

/* MPU6050 手势检测任务 (周期 500ms) */
void vTask_GestureDetect(void *param) {
    int16_t ax, ay, az;
    float threshold = 0.5f;  /* 运动阈值 (g) */

    for (;;) {
        MPU6050_ReadAccel(&ax, &ay, &az);

        float fx = ax / 16384.0f;
        float fy = ay / 16384.0f;
        float fz = az / 16384.0f;

        float motion = sqrtf(fx*fx + fy*fy + fz*fz) - 1.0f;
        if (motion < 0) motion = -motion;

        if (motion > threshold) {
            /* 检测到运动 */
            idle_seconds = 0;
            if (state == WORK_SLEEP) {
                state = WORK_BOOST;   /* 唤醒 */
            }
        } else {
            if (state == WORK_NORMAL) {
                idle_seconds++;
                if (idle_seconds >= 360) {   /* 3 分钟 */
                    state = WORK_SLEEP;
                }
            }
        }

        vTaskDelay(pdMS_TO_TICKS(500));
    }
}
