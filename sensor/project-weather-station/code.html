核心固件代码

数据采集流程：每 5 分钟采集一轮所有传感器数据，校验后通过 LoRa 发送至网关并写入 SD 卡。
/* 环境监测数据采集主循环 */
void EnvMonitor_Task(void) {
    EnvData_t data;

    /* 1. 读取温湿度 */
    DHT22_Read(&data.temp, &data.humi);

    /* 2. 读取气压 */
    BMP280_Read(&data.pressure, &data.altitude);

    /* 3. 读取 PM2.5/PM10 */
    PMS5003_Read(&data.pm25, &data.pm10);
    if (data.pm25 == 0) data.pm25 = PMS5003_GetLast(); // 容错

    /* 4. 风速（每 5s 脉冲计数换算） */
    uint32_t pulse = TIM_GetCapture(TIM3);
    data.wind_speed = pulse * WIND_CALIB_FACTOR;
    TIM_SetCounter(TIM3, 0);

    /* 5. 风向（ADC 采样） */
    uint16_t adc = ADC_Read(ADC_CH_WIND_DIR);
    data.wind_dir = ADC_To_Direction(adc);

    /* 6. 紫外线指数 */
    adc = ADC_Read(ADC_CH_UVI);
    data.uvi = adc * 0.1f;

    /* 7. 时间戳（RTC） */
    data.timestamp = RTC_GetUnixTime();

    /* 8. 写入 SD 卡（CSV 格式） */
    char line[128];
    sprintf(line, "%lu,%.1f,%.1f,%.0f,%d,%d,%.1f,%d,%.1f\n",
        data.timestamp, data.temp, data.humi, data.pressure,
        data.pm25, data.pm10, data.wind_speed, data.wind_dir, data.uvi);
    SD_AppendFile("data.csv", line);

    /* 9. LoRa 发送 */
    LoRa_Send((uint8_t*)&data, sizeof(data));
}
