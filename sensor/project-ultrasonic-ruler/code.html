核心代码 - 超声波测距与温度补偿

测距原理：Trig 脚输出 10us TTL 高电平触发测距，Echo 脚返回高电平脉宽与距离成正比。 利用 NTC 温度传感器实时采集环境温度，对声速进行温度补偿，显著提高测量精度。 
/* 超声波测距仪核心代码 - 温补测距 */

/* 温度补偿声速: v = 331.3 * sqrt(1 + T/273.15) */
float GetSoundSpeed(float temp_celsius) {
    return 331.3f * sqrtf(1.0f + temp_celsius / 273.15f);
}

/* 读取温度 (DS18B20 或 NTC) */
float Read_Temperature(void) {
    uint32_t adc = 0;
    for (int i = 0; i < 8; i++) {
        adc += HAL_ADC_GetValue(&hadc1);
    }
    adc >>= 3; /* 均值滤波 */
    float R_ntc = 10000.0f / (4095.0f / adc - 1.0f);
    /* NTC 10K 3950 B 值 */
    float temp = 1.0f / (logf(R_ntc / 10000.0f) / 3950.0f + 1.0f / 298.15f) - 273.15f;
    return temp;
}

/* 精密度距 (带温度补偿) */
float Ultrasonic_Measure(void) {
    /* 1. 触发测距 */
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_0, GPIO_PIN_SET);
    delay_us(12);
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_0, GPIO_PIN_RESET);

    /* 2. 等待 Echo 高电平 */
    uint32_t timeout = 0;
    while (!HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_1)) {
        if (++timeout > 50000) return -1.0f;
    }

    /* 3. 测量高电平宽度 */
    uint32_t start = micros();
    timeout = 0;
    while (HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_1)) {
        if (++timeout > 50000) return -1.0f;
    }
    uint32_t duration = micros() - start;

    /* 4. 温度补偿声速 */
    float temp = Read_Temperature();
    float sound_speed = GetSoundSpeed(temp); /* mm/us */

    /* 5. 计算距离 (往返折半) */
    float distance_mm = duration * sound_speed / 2.0f;

    /* 6. 多次测量中值滤波 */
    static float buf[5] = {0};
    static uint8_t idx = 0;
    buf[idx++ % 5] = distance_mm;

    /* 冒泡取中值 */
    float sorted[5];
    memcpy(sorted, buf, sizeof(sorted));
    for (int i = 0; i < 5; i++) {
        for (int j = i + 1; j < 5; j++) {
            if (sorted[i] > sorted[j]) {
                float tmp = sorted[i];
                sorted[i] = sorted[j];
                sorted[j] = tmp;
            }
        }
    }
    return sorted[2];
}

/* 蜂鸣器声音提示 - PWM 频率随距离变化 */
void Buzzer_Beep(float distance_cm) {
    uint16_t freq = 0;
    if (distance_cm < 10) {
        freq = 4000;   /* 急促高音 */
    } else if (distance_cm < 50) {
        freq = 2000;   /* 中音 */
    } else if (distance_cm < 100) {
        freq = 1000;   /* 低音 */
    } else {
        freq = 0;      /* 静音 */
    }
    if (freq > 0) {
        __HAL_TIM_SET_AUTORELOAD(&htim3, 72000000 / freq - 1);
        __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_3, 36000000 / freq - 1);
        HAL_TIM_PWM_Start(&htim3, TIM_CHANNEL_3);
    } else {
        HAL_TIM_PWM_Stop(&htim3, TIM_CHANNEL_3);
    }
}

/* 蓝牙发送测量数据 */
void BT_SendData(float distance, float temp, float max_val, float min_val) {
    char buf[64];
    sprintf(buf, "DIST:%.2f,TEMP:%.1f,MAX:%.2f,MIN:%.2f\r\n",
            distance, temp, max_val, min_val);
    HAL_UART_Transmit(&huart2, (uint8_t*)buf, strlen(buf), 100);
}
