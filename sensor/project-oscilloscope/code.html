/* 示波器 ADC 采样 + DMA + 波形绘制 */

#define SAMPLE_DEPTH  256   // 采样深度
uint16_t adc_buf[SAMPLE_DEPTH * 2];  // CH1 + CH2 交替

/* 定时器触发 ADC 采样（DMA 自动搬运） */
void ADC_Init_ForScope(void) {
    /* TIM3 输出 TRGO 触发 ADC */
    TIM_TimeBaseInitTypeDef tim = {
        .TIM_Period = 72 - 1,       // 72MHz/72 = 1MHz → 1us 采样间隔
        .TIM_Prescaler = 1 - 1,
    };
    TIM_TimeBaseInit(TIM3, &tim);
    TIM_SelectOutputTrigger(TIM3, TIM_TRGOSource_Update);

    /* ADC1 双通道扫描 + DMA */
    ADC_InitTypeDef adc = {
        .ADC_Mode = ADC_Mode_Independent,
        .ADC_ScanConvMode = ENABLE,
        .ADC_ContinuousConvMode = ENABLE,
        .ADC_ExternalTrigConv = ADC_ExternalTrigConv_T3_TRGO,
    };
    ADC_Init(ADC1, &adc);
    ADC_RegularChannelConfig(ADC1, ADC_Channel_0, 1, ADC_SampleTime_13Cycles5);
    ADC_RegularChannelConfig(ADC1, ADC_Channel_1, 2, ADC_SampleTime_13Cycles5);

    DMA_InitTypeDef dma = {
        .DMA_PeripheralBaseAddr = (uint32_t)&ADC1->DR,
        .DMA_MemoryBaseAddr = (uint32_t)adc_buf,
        .DMA_BufferSize = SAMPLE_DEPTH * 2,
        .DMA_Mode = DMA_Mode_Circular,
    };
    DMA_Init(DMA1_Channel1, &dma);
}

/* 波形绘制到 TFT */
void Scope_DrawWaveform(void) {
    static uint16_t prev_y[2] = {0, 0};

    /* 清空波形区域 */
    TFT_FillRect(40, 10, 280, 220, COLOR_BLACK);

    /* 绘制网格 */
    for (int i = 0; i < 10; i++)
        TFT_DrawLine(40, 10 + i*22, 320, 10 + i*22, COLOR_GRID);

    /* 绘制 CH1（黄色）*/
    for (int i = 1; i < SAMPLE_DEPTH; i++) {
        uint16_t y1 = 10 + (adc_buf[i-1] * 220 / 4096);
        uint16_t y2 = 10 + (adc_buf[i]   * 220 / 4096);
        TFT_DrawLine(40 + i, y1, 40 + i + 1, y2, COLOR_YELLOW);
    }

    /* 绘制 CH2（蓝色）*/
    for (int i = 1; i < SAMPLE_DEPTH; i++) {
        uint16_t y1 = 10 + (adc_buf[SAMPLE_DEPTH + i - 1] * 220 / 4096);
        uint16_t y2 = 10 + (adc_buf[SAMPLE_DEPTH + i]     * 220 / 4096);
        TFT_DrawLine(40 + i, y1, 40 + i + 1, y2, COLOR_CYAN);
    }

    /* 触发位置标记 */
    TFT_DrawLine(40 + trigger_pos, 10, 40 + trigger_pos, 230, COLOR_RED);
}
