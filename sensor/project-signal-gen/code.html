核心代码 - AD9833 DDS 驱动

驱动说明：AD9833 通过 SPI 接口接收 16bit 控制寄存器配置，设置频率寄存器和相位寄存器即可输出指定波形。以下代码演示了初始化、频率设置和波形切换的核心函数。 
/* AD9833 DDS 驱动核心代码 */

#define AD9833_SPI  hspi1

/* AD9833 寄存器地址 */
#define AD9833_CMD_RESET    0x0100
#define AD9833_CMD_SINE     0x2000
#define AD9833_CMD_SQUARE   0x2028
#define AD9833_CMD_TRIANGLE 0x2002
#define AD9833_CMD_FREQ0    0x4000
#define AD9833_CMD_PHASE0   0xC000

/* 主时钟频率 25MHz */
#define MCLK 25000000.0f

/* 写 16bit 到 AD9833 */
static void AD9833_Write(uint16_t data) {
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_4, GPIO_PIN_RESET); /* FSYNC = 0 */
    HAL_SPI_Transmit(&AD9833_SPI, (uint8_t*)&data, 1, 10);
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_4, GPIO_PIN_SET);   /* FSYNC = 1 */
}

/* 初始化 AD9833 */
void AD9833_Init(void) {
    AD9833_Write(AD9833_CMD_RESET);      /* 复位 */
    AD9833_Write(AD9833_CMD_FREQ0 | 0);  /* FREQ0 REG LSB */
    AD9833_Write(AD9833_CMD_FREQ0 | 0);  /* FREQ0 REG MSB */
    AD9833_Write(AD9833_CMD_PHASE0 | 0); /* PHASE0 REG */
    AD9833_Write(AD9833_CMD_RESET);      /* 退出复位 */
    AD9833_Write(AD9833_CMD_SINE);       /* 默认正弦波 */
}

/* 设置输出频率 (Hz) */
void AD9833_SetFrequency(float freq_hz) {
    /* 计算频率字: FREQ = (f_out / MCLK) * 2^28 */
    uint32_t freq_word = (uint32_t)((freq_hz / MCLK) * 268435456.0f);

    /* 写入 FREQ0 寄存器 (分两次 14bit) */
    uint16_t lsb = (uint16_t)(freq_word & 0x3FFF);
    uint16_t msb = (uint16_t)((freq_word >> 14) & 0x3FFF);

    AD9833_Write(AD9833_CMD_FREQ0 | lsb);
    AD9833_Write(AD9833_CMD_FREQ0 | msb);
}

/* 设置输出波形类型 */
typedef enum {
    WAVE_SINE     = 0,
    WAVE_SQUARE   = 1,
    WAVE_TRIANGLE = 2
} WaveType_t;

void AD9833_SetWaveform(WaveType_t wave) {
    switch (wave) {
        case WAVE_SINE:
            AD9833_Write(AD9833_CMD_SINE);
            break;
        case WAVE_SQUARE:
            AD9833_Write(AD9833_CMD_SQUARE);
            break;
        case WAVE_TRIANGLE:
            AD9833_Write(AD9833_CMD_TRIANGLE);
            break;
    }
}

/* 扫频模式：从 f_start 到 f_stop，step 步进，delay_ms 步进间隔 */
void AD9833_Sweep(float f_start, float f_stop,
                  float step, uint32_t delay_ms) {
    float f = f_start;
    int8_t dir = (f_stop > f_start) ? 1 : -1;
    while ((dir > 0) ? (f <= f_stop) : (f >= f_stop)) {
        AD9833_SetFrequency(f);
        HAL_Delay(delay_ms);
        f += step * dir;
    }
}
