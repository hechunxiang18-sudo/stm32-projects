核心代码 - PI 恒流控制算法

控制算法说明：采用 PI 闭环控制算法，通过 ADC 采样实际电流与设定电流的偏差，调整 MCP4725 DAC 输出电压以控制 MOS 管导通程度，实现高精度恒流放电。 
/* 电子负载仪 PI 恒流控制核心 */

/* PI 控制器参数 */
typedef struct {
    float Kp;       /* 比例系数 */
    float Ki;       /* 积分系数 */
    float integral; /* 积分累加 */
    float limit;    /* 输出限幅 */
    float last_err; /* 上次偏差 */
} PID_t;

PID_t current_pid = {
    .Kp = 1.5f,
    .Ki = 0.2f,
    .integral = 0,
    .limit = 3.3f,   /* DAC 输出范围 0-3.3V */
    .last_err = 0
};

/* 目标电流 (A) */
float target_current = 1.0f;

/* PI 控制器计算 */
float PID_Update(PID_t *pid, float setpoint, float measurement) {
    float error = setpoint - measurement;

    /* 比例项 */
    float p_out = pid->Kp * error;

    /* 积分项（带抗积分饱和） */
    pid->integral += error;
    if (pid->integral > pid->limit / pid->Ki)
        pid->integral = pid->limit / pid->Ki;
    else if (pid->integral < -pid->limit / pid->Ki)
        pid->integral = -pid->limit / pid->Ki;
    float i_out = pid->Ki * pid->integral;

    float output = p_out + i_out;

    /* 输出限幅 */
    if (output > pid->limit) output = pid->limit;
    else if (output < 0) output = 0;

    pid->last_err = error;
    return output;
}

/* ADC 读取电流值 (A) */
float Get_Current(void) {
    uint32_t adc_val = 0;
    for (int i = 0; i < 16; i++) {
        adc_val += HAL_ADC_GetValue(&hadc1);
    }
    adc_val >>= 4; /* 16 次均值滤波 */

    /* 采样电阻 0.1Ω，运放增益 10 倍 */
    float voltage = (adc_val / 4095.0f) * 3.3f;
    return voltage / (10.0f * 0.1f); /* I = V/(Gain * Rsense) */
}

/* 设置 DAC 输出电压 */
void DAC_SetOutput(float voltage) {
    /* MCP4725 12bit DAC */
    uint16_t dac_val = (uint16_t)(voltage / 3.3f * 4095);
    MCP4725_Write(dac_val);
}

/* 恒流控制主循环 (1kHz) */
void CC_ControlLoop(void) {
    float actual_current = Get_Current();
    float control = PID_Update(¤t_pid,
                                target_current, actual_current);
    DAC_SetOutput(control);

    /* 计算实时功率和容量 */
    float voltage = Get_Voltage();
    float power = actual_current * voltage;
    capacity_mAh += actual_current * (1.0f / 3600.0f); /* 每秒累加 */

    /* 更新显示 */
    OLED_ShowData(actual_current, voltage, power, capacity_mAh);

    /* 温度检测 - 过温保护 */
    float temp = Get_Temperature();
    if (temp > 80.0f) {
        DAC_SetOutput(0);         /* 紧急关断 */
        OLED_ShowAlert("OVERTEMP!");
        Fan_SetSpeed(100);        /* 风扇全速 */
    } else if (temp > 50.0f) {
        Fan_SetSpeed((uint8_t)((temp - 50) / 30 * 100)); /* 自动调速 */
    } else {
        Fan_SetSpeed(0);          /* 风扇停止 */
    }

    /* 截止电压检测 */
    if (voltage < cutoff_voltage) {
        DAC_SetOutput(0);
        OLED_ShowAlert("CUTOFF!");
    }
}
