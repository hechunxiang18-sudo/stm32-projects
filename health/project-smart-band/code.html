/* 智能手环核心算法：计步 + 心率 */

/* 计步算法（峰值阈值法）*/
static int step_count = 0;
static float prev_accel_mag = 0;
static uint8_t step_state = 0;  // 0=等待峰值, 1=等待谷值

void StepCounter_Update(float ax, float ay, float az) {
    /* 合成加速度幅度 */
    float mag = sqrtf(ax*ax + ay*ay + az*az);
    const float THRESH_HIGH = 1.3f;  // 峰值阈值
    const float THRESH_LOW  = 0.7f;  // 谷值阈值

    if (step_state == 0) {
        if (mag > THRESH_HIGH && prev_accel_mag <= THRESH_HIGH) {
            step_state = 1;
            step_count++;
        }
    } else {
        if (mag < THRESH_LOW) step_state = 0;
    }
    prev_accel_mag = mag;
}

/* 心率算法（PPG 峰值间隔法）*/
#define PPG_SAMPLE_RATE  50   // 50Hz
#define MIN_BPM  40
#define MAX_BPM  200

uint16_t HeartRate_Calculate(uint16_t ppg_value) {
    static uint16_t buffer[300];  // 6 秒数据窗口
    static uint8_t index = 0;
    static uint32_t last_peak = 0;

    buffer[index++] = ppg_value;
    if (index >= 300) index = 0;

    /* DC 滤波 + 峰值检测 */
    uint32_t sum = 0;
    for (int i = 0; i < 300; i++) sum += buffer[i];
    uint16_t dc = sum / 300;

    /* AC 耦合后查找峰值 */
    static uint8_t peak_detected = 0;
    if (ppg_value > dc + 200 && !peak_detected) {
        peak_detected = 1;
        uint32_t now = HAL_GetTick();

        if (last_peak > 0) {
            uint32_t interval = now - last_peak;  // ms
            uint16_t bpm = 60000 / interval;
            if (bpm >= MIN_BPM && bpm <= MAX_BPM)
                return bpm;
        }
        last_peak = now;
    }

    if (ppg_value < dc) peak_detected = 0;
    return 0;  // 未稳定
}
