核心固件代码

/* 智能药盒主控逻辑 */

typedef struct {
    uint8_t hour;
    uint8_t minute;
    uint8_t slot_id;     // 药仓号 (0-5)
    uint8_t enabled;
} Alarm_t;

Alarm_t alarms[6];

/* RTC 闹钟中断回调 */
void HAL_RTC_AlarmAEventCallback(RTC_HandleTypeDef *hrtc) {
    /* 获取当前时间，匹配闹钟 */
    RTC_TimeTypeDef now;
    HAL_RTC_GetTime(hrtc, &now, RTC_FORMAT_BIN);

    for (int i = 0; i < 6; i++) {
        if (!alarms[i].enabled) continue;
        if (alarms[i].hour == now.Hours && alarms[i].minute == now.Minutes) {
            PillBox_Execute(i);
            break;
        }
    }
}

void PillBox_Execute(uint8_t alarm_idx) {
    OLED_ShowAlarm(alarm_idx);

    // 1. 语音播报
    JQ8900_PlayTrack(alarm_idx);
    delay_ms(200);

    // 2. 驱动步进电机旋转至对应药仓
    Stepper_RotateToSlot(alarms[alarm_idx].slot_id);

    // 3. 等待取药（最多 5 分钟）
    uint32_t start = HAL_GetTick();
    uint8_t taken = 0;
    while ((HAL_GetTick() - start) < 300000) {  // 5min
        if (IR_Sensor_Check()) {                  // 手伸入检测
            taken = 1;
            break;
        }
        delay_ms(50);
    }

    // 4. 取药结果处理
    if (taken) {
        ESP8266_SendMsg("服药提醒", "已按时服药");
        OLED_ShowIcon(ICON_OK);
    } else {
        ESP8266_SendMsg("服药提醒", "未取药！请关注！");
        OLED_ShowIcon(ICON_WARN);
        Buzzer_Alert();  // 持续蜂鸣
    }

    Stepper_ReturnHome();  // 药仓归位
}
