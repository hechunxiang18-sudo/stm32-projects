核心代码 - HX711 称重模块

代码说明：HX711 驱动采用 GPIO 模拟时序，读取 24 位 ADC 值后转换为实际重量。 支持软件滤波（中位值平均滤波）以及去皮/置零功能。 
/* HX711 驱动 - 高精度称重读取 (HAL 库) */

#define HX711_SCK   GPIO_PIN_4  /* PA4 */
#define HX711_DOUT  GPIO_PIN_5  /* PA5 */
#define HX711_PORT  GPIOA

static int32_t offset = 0;      /* 去皮零位偏移 */
static float    scale  = 420.0f;/* 校准系数 */

/* HX711 复位 */
void HX711_Reset(void) {
    HAL_GPIO_WritePin(HX711_PORT, HX711_SCK, GPIO_PIN_SET);
    HAL_Delay(1);
    HAL_GPIO_WritePin(HX711_PORT, HX711_SCK, GPIO_PIN_RESET);
    HAL_Delay(1);
}

/* 读取 HX711 原始 24 位数据 */
int32_t HX711_ReadRaw(void) {
    int32_t val = 0;
    while (HAL_GPIO_ReadPin(HX711_PORT, HX711_DOUT) == GPIO_PIN_SET);
    for (int i = 0; i < 24; i++) {
        HAL_GPIO_WritePin(HX711_PORT, HX711_SCK, GPIO_PIN_SET);
        val = (val << 1) | HAL_GPIO_ReadPin(HX711_PORT, HX711_DOUT);
        HAL_GPIO_WritePin(HX711_PORT, HX711_SCK, GPIO_PIN_RESET);
    }
    /* 第 25 个脉冲设置增益 (128) */
    HAL_GPIO_WritePin(HX711_PORT, HX711_SCK, GPIO_PIN_SET);
    HAL_Delay_us(1);
    HAL_GPIO_WritePin(HX711_PORT, HX711_SCK, GPIO_PIN_RESET);
    return (val & 0x800000) ? (val | 0xFF000000) : val;
}

/* 软件滤波 - 中位值平均滤波 */
#define FILTER_N  5
float HX711_Filter(void) {
    int32_t buf[FILTER_N], tmp;
    for (int i = 0; i < FILTER_N; i++) {
        buf[i] = HX711_ReadRaw();
        HAL_Delay(10);
    }
    /* 排序 */
    for (int i = 0; i < FILTER_N - 1; i++)
        for (int j = i + 1; j < FILTER_N; j++)
            if (buf[i] > buf[j]) {
                tmp = buf[i]; buf[i] = buf[j]; buf[j] = tmp;
            }
    /* 去除最大最小，取平均 */
    int32_t sum = 0;
    for (int i = 1; i < FILTER_N - 1; i++) sum += buf[i];
    return (float)(sum) / (FILTER_N - 2);
}

/* 获取重量 (g) */
float HX711_GetWeight(void) {
    float raw = HX711_Filter();
    return (raw - offset) / scale;
}

/* 去皮 */
void HX711_Tare(void) {
    offset = (int32_t)HX711_Filter();
}

/* 置零 */
void HX711_Zero(void) {
    offset = 0;
}
