核心代码 - 电化学检测与浓度计算

代码说明：恒电位仪施加 0.4V 工作电压，等待反应稳定后采样电流值，通过标定曲线计算血糖浓度。 
/* 智能血糖仪 - 电化学检测与浓度计算 */

#include "glucometer.h"
#include "ads1115.h"
#include "potentiostat.h"
#include "eeprom_sim.h"

/* 标定参数 (存储在 EEPROM) */
typedef struct {
    float slope;       /* 斜率 (μA / mmol/L) */
    float intercept;   /* 截距 (μA) */
    float temp_coeff;  /* 温度补偿系数 */
} CalibrationParams;

static CalibrationParams cal;

/* 检测流程状态机 */
typedef enum {
    IDLE,
    STRIP_INSERTED,
    BLOOD_DETECTED,
    REACTION_IN_PROGRESS,
    MEASUREMENT_DONE,
    ERROR
} MeasState;

static MeasState state = IDLE;

/* 试纸插入检测 */
uint8_t glucometer_detect_strip(void) {
    /* PA0 为高表示试纸插入 */
    if (GPIO_ReadPin(GPIOA, GPIO_PIN_0)) {
        /* 检测试纸电阻识别类型 */
        uint16_t resistance = ads1115_read_single(ADS1115_CH1);
        if (resistance > 100 && resistance < 4000) {
            return 1;  /* 有效试纸 */
        }
    }
    return 0;
}

/* 血样检测 - 检测电极间阻抗变化 */
uint8_t glucometer_detect_blood(void) {
    potentiostat_set_voltage(0.4f);   /* 施加 0.4V */
    delay_ms(100);
    uint32_t sample = 0;
    for (int i = 0; i < 10; i++) {
        sample += ads1115_read_single(ADS1115_CH0);
        delay_ms(2);
    }
    sample /= 10;

    /* 血样使电流突增，超过阈值判定为有血样 */
    return (sample > BLOOD_THRESHOLD) ? 1 : 0;
}

/* 葡萄糖浓度计算 (温度补偿) */
float glucometer_calc_glucose(uint16_t adc_raw, float temp_c) {
    /* ADC 值 → 电流 (μA) */
    float current_ua = (adc_raw / 32768.0f) * 4.096f / 100.0f * 1e6f;

    /* 温度补偿 */
    float temp_factor = 1.0f + cal.temp_coeff * (temp_c - 25.0f);
    current_ua /= temp_factor;

    /* 浓度计算: C = (I - intercept) / slope */
    float glucose = (current_ua - cal.intercept) / cal.slope;

    if (glucose < 0)   glucose = 0;
    if (glucose > 33.3) glucose = 33.3;  /* 量程上限 */

    return glucose;
}

/* 完整的测量流程 */
float glucometer_measure(void) {
    state = STRIP_INSERTED;

    if (!glucometer_detect_strip()) {
        state = ERROR;
        return -1;
    }
    OLED_ShowMessage("试纸已识别");

    /* 等待血样 */
    uint32_t wait_start = HAL_GetTick();
    while (!glucometer_detect_blood()) {
        if (HAL_GetTick() - wait_start > 10000) {
            state = ERROR;
            return -2;  /* 超时 */
        }
        delay_ms(50);
    }
    state = BLOOD_DETECTED;
    OLED_ShowMessage("血样检测中...");

    /* 反应 5 秒后读取 */
    state = REACTION_IN_PROGRESS;
    delay_ms(5000);

    /* 多次采样取平均 */
    uint32_t sum = 0;
    for (int i = 0; i < 20; i++) {
        sum += ads1115_read_single(ADS1115_CH0);
        delay_ms(10);
    }
    float temp = ds18b20_read_temp();  /* 温度补偿 */
    float glucose = glucometer_calc_glucose(sum / 20, temp);

    state = MEASUREMENT_DONE;
    return glucose;
}
