核心软件代码

代码说明：以下展示避障小车的核心代码，包括多路超声波测距、舵机扫描、路径规划算法、电机控制与 OLED 雷达显示。 
/* ========== 避障小车 - 路径规划主程序 ========== */

#include "hcsr04.h"
#include "servo.h"
#include "motor.h"
#include "oled.h"

#define SPEED_SLOW   400    /* 慢速 */
#define SPEED_NORMAL 800    /* 正常 */
#define SPEED_FAST   1200   /* 快速 */

/* 环境感知数据结构 */
typedef struct {
    uint16_t frontDist;     /* 前方距离 (cm) */
    uint16_t leftDist;      /* 左侧距离 (cm) */
    uint16_t rightDist;     /* 右侧距离 (cm) */
    uint16_t scanMap[180];  /* 舵机扫描地图 (角度->距离) */
} Environment_t;

Environment_t g_env;
volatile uint8_t g_mode = MODE_AUTO;  /* MODE_AUTO / MODE_BT */

/* ===== 1. 多路超声波测距 ===== */
uint16_t Ultrasonic_Measure(uint8_t trigPin, uint8_t echoPin) {
    uint32_t tick;
    GPIO_WriteBit(GPIOA, trigPin, SET);
    delay_us(10);
    GPIO_WriteBit(GPIOA, trigPin, RESET);

    uint32_t timeout = 0;
    while (GPIO_ReadInputDataBit(GPIOA, echoPin) == 0) {
        if (++timeout > 3000) return 999;
        delay_us(10);
    }

    tick = GetSysTick();
    timeout = 0;
    while (GPIO_ReadInputDataBit(GPIOA, echoPin) == 1) {
        if (++timeout > 30000) return 999;
        delay_us(10);
    }

    return (GetSysTick() - tick) / 58;
}

void Environment_Update(void) {
    g_env.frontDist = Ultrasonic_Measure(GPIO_Pin_0, GPIO_Pin_1);
    g_env.leftDist  = Ultrasonic_Measure(GPIO_Pin_2, GPIO_Pin_3);
    g_env.rightDist = Ultrasonic_Measure(GPIO_Pin_4, GPIO_Pin_5);
}

/* ===== 2. 舵机扫描（180 度建图） ===== */
void Servo_Scan(void) {
    /* 从 0 度到 180 度，每 2 度采一次 */
    for (uint8_t angle = 0; angle <= 180; angle += 2) {
        /* 设置舵机角度 */
        Servo_SetAngle(angle);
        HAL_Delay(10);  /* 等待舵机到位 */

        /* 测量距离 */
        uint16_t dist = Ultrasonic_Measure(GPIO_Pin_0, GPIO_Pin_1);
        g_env.scanMap[angle] = dist;

        /* 更新 OLED 雷达图（每 5 度刷新） */
        if (angle % 10 == 0) {
            OLED_DrawRadar(g_env.scanMap, angle);
        }
    }
}

/* ===== 3. 路径规划算法 ===== */
typedef enum { FORWARD, TURN_LEFT, TURN_RIGHT, BACK_OFF, STOP } Action_t;

Action_t Path_Planning(void) {
    uint16_t f = g_env.frontDist;
    uint16_t l = g_env.leftDist;
    uint16_t r = g_env.rightDist;

    /* 安全距离阈值 */
    #define SAFE_DIST     30   /* 安全距离 (cm) */
    #define DANGER_DIST   15   /* 危险距离 (cm) */

    /* --- 决策树 --- */

    /* 前方有障碍 */
    if (f < SAFE_DIST) {
        if (f < DANGER_DIST) {
            /* 非常靠近障碍 -> 后退 */
            return BACK_OFF;
        }

        /* 判断左右哪边空间更大 */
        if (l > r && l > SAFE_DIST) {
            return TURN_LEFT;
        } else if (r > l && r > SAFE_DIST) {
            return TURN_RIGHT;
        } else {
            /* 左右都有障碍 -> 后退 */
            return BACK_OFF;
        }
    }

    /* 前方无障碍正常前进，但检测是否需要微调方向 */
    if (l < SAFE_DIST && r > SAFE_DIST) {
        /* 左侧有障碍 -> 向右微调 */
        return TURN_RIGHT;
    }
    if (r < SAFE_DIST && l > SAFE_DIST) {
        /* 右侧有障碍 -> 向左微调 */
        return TURN_LEFT;
    }

    /* 前方及两侧都安全 -> 直行 */
    return FORWARD;
}

/* ===== 4. 电机执行动作 ===== */
void Motor_Execute(Action_t action) {
    switch (action) {
        case FORWARD:
            Motor_SetSpeed(SPEED_NORMAL, SPEED_NORMAL);
            Motor_SetDirection(FORWARD, FORWARD);
            break;

        case TURN_LEFT:
            Motor_SetSpeed(SPEED_SLOW, SPEED_NORMAL);
            Motor_SetDirection(FORWARD, FORWARD);
            break;

        case TURN_RIGHT:
            Motor_SetSpeed(SPEED_NORMAL, SPEED_SLOW);
            Motor_SetDirection(FORWARD, FORWARD);
            break;

        case BACK_OFF:
            Motor_SetSpeed(SPEED_NORMAL, SPEED_NORMAL);
            Motor_SetDirection(BACKWARD, BACKWARD);
            HAL_Delay(500);  /* 后退 0.5 秒 */
            /* 随机转向 */
            if (g_env.leftDist > g_env.rightDist) {
                Motor_SetSpeed(SPEED_SLOW, 0);
            } else {
                Motor_SetSpeed(0, SPEED_SLOW);
            }
            HAL_Delay(300);
            break;

        case STOP:
            Motor_Stop();
            break;
    }
}

/* ===== 5. 速度自适应 ===== */
uint16_t Speed_Adaptive(void) {
    uint16_t minDist = g_env.frontDist;
    if (g_env.leftDist < minDist) minDist = g_env.leftDist;
    if (g_env.rightDist < minDist) minDist = g_env.rightDist;

    /* 根据最近障碍物距离自动调整速度 */
    if (minDist > 100) {
        return SPEED_FAST;    /* 开阔区域加速 */
    } else if (minDist > 50) {
        return SPEED_NORMAL;  /* 正常距离 */
    } else if (minDist > 20) {
        return SPEED_SLOW;    /* 接近障碍减速 */
    } else {
        return 0;             /* 停止 */
    }
}

/* ===== 6. OLED 雷达图绘制 ===== */
void OLED_DrawRadar(uint16_t *scanMap, uint8_t maxAngle) {
    /* 简单极坐标转直角坐标绘制 */
    /* 128x64 OLED, 中心在 (64, 56), 半径 40px */
    const int16_t cx = 64, cy = 56, rMax = 40;

    OLED_Clear();

    /* 绘制距离标尺 */
    OLED_DrawCircle(cx, cy, 10);  /* 25cm */
    OLED_DrawCircle(cx, cy, 20);  /* 50cm */
    OLED_DrawCircle(cx, cy, 30);  /* 75cm */
    OLED_DrawCircle(cx, cy, rMax);/* 100cm */

    /* 绘制扫描点 */
    for (uint8_t a = 0; a <= maxAngle; a += 2) {
        uint16_t dist = scanMap[a];
        if (dist > 100) continue;  /* 超出量程 */

        /* 距离映射到半径 */
        uint8_t r = (uint8_t)((float)dist / 100.0f * rMax);
        float rad = (float)a * 3.14159f / 180.0f;
        int16_t x = cx + (int16_t)(r * sinf(rad));
        int16_t y = cy - (int16_t)(r * cosf(rad));

        OLED_DrawPixel(x, y, 1);
    }

    /* 显示前方距离 */
    char buf[8];
    sprintf(buf, "%3dcm", g_env.frontDist);
    OLED_ShowString(0, 0, buf);
}

/* ===== 7. 蓝牙指令处理 ===== */
void Bluetooth_Process(void) {
    if (!BT_DataAvailable()) return;

    char cmd[16];
    BT_Receive((uint8_t *)cmd, 16);

    switch (cmd[0]) {
        case 'W': Motor_SetSpeed(SPEED_NORMAL, SPEED_NORMAL);
                  Motor_SetDirection(FORWARD, FORWARD); break;
        case 'S': Motor_SetSpeed(SPEED_NORMAL, SPEED_NORMAL);
                  Motor_SetDirection(BACKWARD, BACKWARD); break;
        case 'A': Motor_SetSpeed(SPEED_NORMAL, SPEED_NORMAL);
                  Motor_SetDirection(BACKWARD, FORWARD); break;
        case 'D': Motor_SetSpeed(SPEED_NORMAL, SPEED_NORMAL);
                  Motor_SetDirection(FORWARD, BACKWARD); break;
        case 'M': g_mode = MODE_AUTO; break;  /* 切换自动模式 */
        case 'X': Motor_Stop();
                  g_mode = MODE_BT;  break;   /* 切换遥控模式 */
    }
}

/* ===== 8. 主循环 ===== */
int main(void) {
    HAL_Init();
    SystemClock_Config();

    GPIO_Init();             /* 超声波 IO */
    TIM3_PWM_Init(1999, 71); /* 舵机 PWM */
    TIM2_PWM_Init(999, 71);  /* 电机 PWM */
    UART1_Init(9600);        /* HC-05 */
    I2C1_Init();             /* OLED */

    Motor_Init();
    OEM_Init();

    OLED_ShowString(0, 0, "避障小车 v2.0");
    OLED_ShowString(2, 0, "自动模式");
    HAL_Delay(1000);

    uint32_t lastScanTick = 0;
    uint32_t lastMeasureTick = 0;

    while (1) {
        uint32_t now = HAL_GetTick();

        /* 环境测量（每 100ms） */
        if (now - lastMeasureTick > 100) {
            Environment_Update();
            lastMeasureTick = now;
        }

        if (g_mode == MODE_AUTO) {
            /* 自动避障模式 */

            /* 每 5 秒做一次舵机扫描 */
            if (now - lastScanTick > 5000) {
                Servo_Scan();
                lastScanTick = now;
            }

            /* 路径规划决策 */
            Action_t action = Path_Planning();

            /* 速度自适应 */
            uint16_t speed = Speed_Adaptive();

            /* 执行动作 */
            if (speed > 0) {
                Motor_Execute(action);
            } else {
                Motor_Stop();
            }

            /* 更新显示 */
            char buf[16];
            sprintf(buf, "F:%3d L:%3d R:%3d",
                    g_env.frontDist, g_env.leftDist, g_env.rightDist);
            OLED_ShowString(1, 0, buf);
        }

        else if (g_mode == MODE_BT) {
            /* 蓝牙遥控模式 */
            Bluetooth_Process();
            OLED_ShowString(1, 0, "蓝牙遥控模式  ");
        }

        HAL_Delay(50);
    }
}
