核心代码 - 逆运动学解算

代码说明：以下为机械臂逆运动学解算核心函数，输入末端执行器坐标 (x, y, z)，输出五个舵机的目标角度，实现空间点到点的平滑运动。 
/* 5轴机械臂逆运动学解算 */

/* 机械臂 D-H 参数（单位：mm）*/
#define L1  60  /* 底座到肩部高度 */
#define L2  80  /* 大臂长度 */
#define L3  70  /* 小臂长度 */
#define L4  40  /* 腕部偏距 */

/* 舵机角度结构体 */
typedef struct {
    float base;    /* 底盘旋转角度 0-180 */
    float shoulder; /* 肩部角度 0-180 */
    float elbow;    /* 肘部角度 0-180 */
    float wrist;    /* 腕部角度 0-180 */
    float gripper;  /* 夹爪开合 0-180 */
} ServoAngles_t;

/* 逆运动学解算：输入末端坐标 (x,y,z)，输出各关节角度 */
ServoAngles_t IK_Solve(float x, float y, float z) {
    ServoAngles_t angles = {0};

    /* 1. 底盘旋转角 = atan2(y, x) */
    angles.base = atan2f(y, x) * 180.0f / M_PI;
    if (angles.base < 0) angles.base += 180;

    /* 2. 计算在肩部坐标系中的位置 */
    float r = sqrtf(x*x + y*y);       /* 水平距离 */
    float h = z - L1;                 /* 垂直高度（减去底座）*/

    /* 3. 余弦定理求解肩部和肘部角度 */
    float d = sqrtf(r*r + h*h);       /* 肩部到腕部距离 */
    float cos_elbow = (L2*L2 + L3*L3 - d*d) / (2*L2*L3);
    float cos_shoulder = (L2*L2 + d*d - L3*L3) / (2*L2*d);

    /* 边界钳位 */
    if (cos_elbow > 1.0f) cos_elbow = 1.0f;
    if (cos_elbow < -1.0f) cos_elbow = -1.0f;
    if (cos_shoulder > 1.0f) cos_shoulder = 1.0f;
    if (cos_shoulder < -1.0f) cos_shoulder = -1.0f;

    /* 肘部角度（锐角模式）*/
    angles.elbow = acosf(cos_elbow) * 180.0f / M_PI;

    /* 肩部角度（含基座偏置补偿）*/
    float alpha = atan2f(h, r);
    float beta = acosf(cos_shoulder);
    angles.shoulder = (alpha + beta) * 180.0f / M_PI;

    /* 4. 腕部角度保持夹爪水平 */
    angles.wrist = 180.0f - angles.shoulder - angles.elbow;
    if (angles.wrist < 0) angles.wrist = 0;

    /* 5. 夹爪默认闭合 */
    angles.gripper = 90.0f;

    return angles;
}

/* 空间直线插补：从起点到终点平滑移动 */
void IK_MoveLinear(float x1, float y1, float z1,
                   float x2, float y2, float z2, int steps) {
    for (int i = 0; i <= steps; i++) {
        float t = (float)i / steps;
        float x = x1 + (x2 - x1) * t;
        float y = y1 + (y2 - y1) * t;
        float z = z1 + (z2 - z1) * t;
        ServoAngles_t ang = IK_Solve(x, y, z);
        PCA9685_SetServoAll(ang);
        delay(20);  /* 20ms 间隔 */
    }
}
