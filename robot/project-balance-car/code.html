核心固件代码

/* 平衡小车串级 PID 控制（1ms 中断）*/

#define ANGLE_LOOP_KP  25.0f
#define ANGLE_LOOP_KD  0.5f
#define SPEED_LOOP_KP  2.0f
#define SPEED_LOOP_KI  0.1f

static float balance_pid_out = 0;
static float speed_pid_out = 0;

/* 1ms 定时器中断 — 核心平衡控制 */
void TIM_IRQ_Handler(void) {
    static uint16_t speed_counter = 0;

    /* 1. 读取陀螺仪 + 加速度计 */
    MPU6050_Read(&gyro_y, &accel_x);

    /* 2. 互补滤波：angle = 0.98*(angle+gyro*dt) + 0.02*accel */
    angle = 0.98f * (angle + gyro_y * 0.001f)
          + 0.02f * atan2f(accel_x, accel_z) * RAD_TO_DEG;

    /* 3. 角度环 PD 控制（400Hz）*/
    balance_pid_out = ANGLE_LOOP_KP * (angle - 0)
                    + ANGLE_LOOP_KD * gyro_y;

    /* 4. 速度环 PI 控制（50Hz）*/
    speed_counter++;
    if (speed_counter >= 20) {  // 每 20ms
        speed_counter = 0;

        int16_t speed_left  = Encoder_GetCount(ENC_LEFT);
        int16_t speed_right = Encoder_GetCount(ENC_RIGHT);
        float speed_avg = (speed_left + speed_right) / 2.0f;

        /* 速度环积分 */
        speed_integral += SPEED_LOOP_KI * speed_avg * 0.02f;
        speed_integral = CLAMP(speed_integral, -50, 50);

        speed_pid_out = SPEED_LOOP_KP * speed_avg + speed_integral;

        Encoder_Reset(ENC_LEFT);
        Encoder_Reset(ENC_RIGHT);
    }

    /* 5. 合成输出 = 角度环 + 速度环 + 遥控值 */
    float motor_out = balance_pid_out + speed_pid_out + rc_cmd;
    motor_out = CLAMP(motor_out, -100, 100);

    /* 6. 驱动电机（死区补偿）*/
    if (fabs(motor_out) > 5) {
        Motor_SetSpeed(MOTOR_LEFT,  motor_out + motor_out*0.05f);
        Motor_SetSpeed(MOTOR_RIGHT, motor_out - motor_out*0.05f);
    } else {
        Motor_Stop();
    }

    /* 7. 摔倒检测（角度过大自动停机）*/
    if (fabs(angle) > 45.0f) {
        Motor_Stop();
        TIM_ITConfig(TIM_IRQ, DISABLE);  // 关闭中断
        OLED_ShowString("FALL DOWN");
    }
}
