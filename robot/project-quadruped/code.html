/* 四足机器人逆运动学 + 步态控制 */

/* 腿部几何参数（mm） */
#define HIP_LEN   35    // 髋关节长度
#define FEMUR_LEN 50    // 大腿长度
#define TIBIA_LEN 55    // 小腿长度

/* 足端坐标 → 舵机角度解算 */
void IK_Solve(LegPos_t *target, ServoAngles_t *angles) {
    float x = target->x;
    float y = target->y;
    float z = target->z;

    /* 髋关节角度（侧向） */
    angles->hip = atan2(y, x) * RAD_TO_DEG;

    /* 腿平面距离 */
    float d = sqrt(x*x + y*y) - HIP_LEN;
    float L = sqrt(d*d + z*z);

    /* 膝关节角度（余弦定理） */
    float cos_knee = (FEMUR_LEN*FEMUR_LEN + TIBIA_LEN*TIBIA_LEN - L*L)
                    / (2 * FEMUR_LEN * TIBIA_LEN);
    angles->knee = acos(CLAMP(cos_knee, -1, 1)) * RAD_TO_DEG;

    /* 髋关节俯仰角度 */
    float alpha = atan2(z, d);
    float beta = acos((FEMUR_LEN*FEMUR_LEN + L*L - TIBIA_LEN*TIBIA_LEN)
                      / (2 * FEMUR_LEN * L));
    angles->hip_pitch = (alpha + beta) * RAD_TO_DEG;
}

/* 三角步态规划 */
void Gait_Trot(void) {
    static float phase = 0;
    float step_period = 0.8f;    // 步态周期 0.8s
    float step_height = 30.0f;   // 抬腿高度 30mm
    float step_length = 40.0f;   // 步幅 40mm

    /* 相位：[0, 0.5) 摆腿相，[0.5, 1) 支撑相 */
    for (int leg = 0; leg < 4; leg++) {
        float leg_phase = fmod(phase + leg * 0.5f, 1.0f);
        float z_offset, x_offset;

        if (leg_phase < 0.5f) {
            /* 摆腿相：抬腿 + 前伸 */
            float t = leg_phase * 2;
            z_offset = step_height * sin(t * PI);
            x_offset = -step_length * (1 - cos(t * PI)) / 2;
        } else {
            /* 支撑相：落地 + 后蹬 */
            float t = (leg_phase - 0.5f) * 2;
            z_offset = 0;
            x_offset = step_length * t / 2;
        }

        LegPos_t target = {
            .x = HOME_X + x_offset,
            .y = HOME_Y + (leg < 2 ? 1 : -1) * HIP_SPACING,
            .z = HOME_Z + z_offset
        };

        ServoAngles_t angles;
        IK_Solve(&target, &angles);
        PCA9685_SetServo(leg*2, angles.hip);
        PCA9685_SetServo(leg*2+1, angles.knee);
    }

    phase += 0.005f;  // 200Hz 更新
    if (phase >= 1) phase -= 1;
}
