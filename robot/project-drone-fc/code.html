核心固件代码

/* 四轴飞控 PID 自稳主循环（1kHz 定时器中断） */

#define PID_KP_ROLL  2.5f
#define PID_KI_ROLL  0.05f
#define PID_KD_ROLL  0.3f

/* 姿态数据 */
typedef struct {
    float roll, pitch, yaw;       // 解算角度
    float gyro_x, gyro_y, gyro_z; // 角速度
} Attitude_t;

/* PID 控制器 */
void PID_Update(PID_t *pid, float target, float current, float gyro) {
    float error = target - current;

    pid->integral += error * dt;
    pid->integral = CLAMP(pid->integral, -LIMIT_INT, LIMIT_INT);

    float derivative = -gyro;  // 角速度作为微分项（D 项）

    pid->output = pid->kp * error
                + pid->ki * pid->integral
                + pid->kd * derivative;

    pid->output = CLAMP(pid->output, -LIMIT_OUT, LIMIT_OUT);
}

/* 主控制循环（1ms 执行一次）*/
void FlightControl_Loop(void) {
    Attitude_t att;

    /* 1. 读取 MPU6050 */
    MPU6050_ReadGyro(&att.gyro_x, &att.gyro_y, &att.gyro_z);
    MPU6050_ReadAccel(&att.accel_x, &att.accel_y, &att.accel_z);

    /* 2. 互补滤波姿态解算 */
    float dt = 0.001f;
    ComplementaryFilter(&att, dt);

    /* 3. 接收遥控器值 */
    RC_GetChannels(&rc_roll, &rc_pitch, &rc_throttle, &rc_yaw);

    /* 4. 级联 PID 控制 */
    pid_roll.target  = rc_roll * MAX_ANGLE;    // 目标角度
    PID_Update(&pid_roll,  pid_roll.target,  att.roll,  att.gyro_x);
    PID_Update(&pid_pitch, rc_pitch * MAX_ANGLE, att.pitch, att.gyro_y);
    PID_Update(&pid_yaw,   rc_yaw, att.yaw, att.gyro_z);

    /* 5. 混控输出 */
    float m1 = rc_throttle - pid_roll.output - pid_pitch.output - pid_yaw.output;
    float m2 = rc_throttle + pid_roll.output - pid_pitch.output + pid_yaw.output;
    float m3 = rc_throttle + pid_roll.output + pid_pitch.output - pid_yaw.output;
    float m4 = rc_throttle - pid_roll.output + pid_pitch.output + pid_yaw.output;

    Motor_Write(MOTOR1, CLAMP(m1, 1000, 2000));
    Motor_Write(MOTOR2, CLAMP(m2, 1000, 2000));
    Motor_Write(MOTOR3, CLAMP(m3, 1000, 2000));
    Motor_Write(MOTOR4, CLAMP(m4, 1000, 2000));
}
