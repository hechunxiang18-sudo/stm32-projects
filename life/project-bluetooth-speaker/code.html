核心代码 - I2S 音频输出与 RGB 灯效

代码说明：STM32 I2S 外设驱动 TPA3116 功放，定时器 PWM 驱动 RGB 灯效， 通过 FFT 频谱分析实现音乐律动效果。 
/* 蓝牙音箱 - I2S 音频与 RGB 律动 (HAL 库) */

#include "arm_math.h"

/* I2S 音频缓冲区 */
#define AUDIO_BUF_SIZE  512
int16_t audio_tx_buf[AUDIO_BUF_SIZE];
volatile uint8_t audio_half_done = 0;

/* I2S DMA 发送完成回调 */
void HAL_I2S_TxHalfCpltCallback(I2S_HandleTypeDef *hi2s) {
    audio_half_done = 1;
    Process_Audio(audio_tx_buf, AUDIO_BUF_SIZE / 2);
}

void HAL_I2S_TxCpltCallback(I2S_HandleTypeDef *hi2s) {
    audio_half_done = 2;
    Process_Audio(audio_tx_buf + AUDIO_BUF_SIZE / 2, AUDIO_BUF_SIZE / 2);
}

/* FFT 频谱分析 (简化) */
#define FFT_SIZE  64
static float32_t fft_input[FFT_SIZE * 2];
static float32_t fft_output[FFT_SIZE];

void Process_Audio(int16_t *buf, uint16_t len) {
    for (int i = 0; i < FFT_SIZE; i++) {
        fft_input[i * 2]     = (float32_t)buf[i] / 32768.0f;
        fft_input[i * 2 + 1] = 0;
    }

    arm_cfft_f32(&arm_cfft_sR_f32_len64, fft_input, 0, 1);
    arm_cmplx_mag_f32(fft_input, fft_output, FFT_SIZE);

    /* 计算低频能量 (0-10 bin) 用于 RGB 亮度 */
    float bass = 0;
    for (int i = 1; i < 10; i++) bass += fft_output[i];
    bass /= 10.0f;

    /* 映射到 RGB 亮度 */
    uint8_t r = (uint8_t)(bass * 255);
    uint8_t g = (uint8_t)(bass * 128);
    uint8_t b = (uint8_t)(bass * 64);

    __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_3, r);
    __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_4, g);
    __HAL_TIM_SET_COMPARE(&htim4, TIM_CHANNEL_1, b);
}

/* TTP224 触摸按键处理 */
void Handle_Touch(void) {
    uint8_t key = ~GPIO_ReadInputData(GPIOC) & 0x0F;
    static uint8_t last_key = 0;
    if (key && key != last_key) {
        if (key & 0x01) EQ_Cycle();       /* EQ 切换 */
        if (key & 0x02) RGB_ModeNext();   /* 灯效模式 */
        if (key & 0x04) Volume_Up();
        if (key & 0x08) Volume_Down();
    }
    last_key = key;
}
