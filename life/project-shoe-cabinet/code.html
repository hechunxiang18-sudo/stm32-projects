核心代码 - PID 恒温控制

代码说明：采用位置式 PID 算法控制 PTC 加热器，根据 DHT22 实测温度动态调节 PWM 占空比，实现精确恒温。 
/* 智能鞋柜 - PID 恒温控制 */

#include "pid_control.h"
#include "dht22.h"
#include "relay.h"

/* PID 参数 */
typedef struct {
    float kp;
    float ki;
    float kd;
    float integral_limit;
    float output_limit;
    float setpoint;
    float integral;
    float prev_error;
} PIDController;

static PIDController pid;

/* 初始化 PID 参数 */
void pid_init(float setpoint) {
    pid.kp = 2.8f;
    pid.ki = 0.12f;
    pid.kd = 0.35f;
    pid.integral_limit = 50.0f;
    pid.output_limit = 100.0f;
    pid.setpoint = setpoint;
    pid.integral = 0.0f;
    pid.prev_error = 0.0f;
}

/* PID 计算 */
float pid_compute(float measurement) {
    float error = pid.setpoint - measurement;

    /* 比例项 */
    float p_term = pid.kp * error;

    /* 积分项 (带抗积分饱和) */
    pid.integral += error;
    if (pid.integral > pid.integral_limit)
        pid.integral = pid.integral_limit;
    if (pid.integral < -pid.integral_limit)
        pid.integral = -pid.integral_limit;
    float i_term = pid.ki * pid.integral;

    /* 微分项 */
    float d_term = pid.kd * (error - pid.prev_error);
    pid.prev_error = error;

    /* 计算输出 */
    float output = p_term + i_term + d_term;

    /* 限幅 */
    if (output > pid.output_limit)  output = pid.output_limit;
    if (output < 0.0f)              output = 0.0f;

    return output;
}

/* 烘干模式主任务 */
void vTask_Drying(void *param) {
    const float target_temp = 45.0f;   /* 目标温度 45°C */
    const uint16_t total_minutes = 120; /* 总计 2 小时 */

    pid_init(target_temp);
    Relay_On(HEAT_RELAY);   /* 启动 PTC 电源 */
    Fan_SetSpeed(FAN1, 80); /* 风扇中速循环 */

    uint16_t elapsed = 0;

    while (elapsed < total_minutes) {
        /* 读取温湿度 */
        DHT22_ReadData();
        float temp = DHT22_GetTemperature();

        /* PID 计算加热占空比 (通过继电器周期性开关实现) */
        float duty = pid_compute(temp);
        uint32_t on_time = (uint32_t)(duty * 100);   /* ms */
        uint32_t off_time = 10000 - on_time;

        /* PID 输出控制加热 */
        if (duty > 5.0f) {
            Relay_On(HEAT_RELAY);
            delay_ms(on_time);
            Relay_Off(HEAT_RELAY);
            delay_ms(off_time);
        } else {
            Relay_Off(HEAT_RELAY);
            delay_ms(10000);
        }

        /* 超温保护 */
        if (temp > 65.0f) {
            Relay_Off(HEAT_RELAY);
            OLED_ShowWarning("超温保护!");
            Buzzer_Beep(3);
            break;
        }

        /* 门开检测 - 开门暂停加热 */
        if (GPIO_ReadPin(GPIOB, GPIO_PIN_0) == 1) {
            Relay_Off(HEAT_RELAY);
            OLED_ShowMessage("门已打开");
            while (GPIO_ReadPin(GPIOB, GPIO_PIN_0) == 1)
                delay_ms(100);
        }

        OLED_UpdateDisplay();
        elapsed++;
        vTaskDelay(pdMS_TO_TICKS(60000)); /* 每分钟循环 */
    }

    /* 完成 */
    Relay_Off(HEAT_RELAY);
    Fan_SetSpeed(FAN1, 0);
    OLED_ShowMessage("烘干完成");
}
