核心代码 - 自动开盖与满桶检测

控制逻辑说明：主循环中同时检测红外信号和超声波距离，实现自动开盖和满桶提示。PIR 触发后开盖计时器启动，人员离开 3 秒后自动关闭。 
/* 智能垃圾桶主控制程序 */

#define TRIG_PIN   GPIO_PIN_1   /* PA1 */
#define ECHO_PIN   GPIO_PIN_2   /* PA2 */
#define PIR_PIN    GPIO_PIN_0   /* PA0 */

#define IR_THRESHOLD  30   /* 感应到人体阈值 */
#define FULL_THRESHOLD 15  /* 满桶距离阈值 (cm) */
#define LID_CLOSE_DELAY 3000 /* 关门延时 ms */

/* 超声波测距 (cm) */
float HC_SR04_GetDistance(void) {
    /* 发送 10us 触发脉冲 */
    HAL_GPIO_WritePin(GPIOA, TRIG_PIN, GPIO_PIN_SET);
    delay_us(10);
    HAL_GPIO_WritePin(GPIOA, TRIG_PIN, GPIO_PIN_RESET);

    /* 等待回波，测量高电平时间 */
    uint32_t timeout = 0;
    while (!HAL_GPIO_ReadPin(GPIOA, ECHO_PIN)) {
        if (++timeout > 100000) return -1;
    }

    uint32_t start = micros();
    timeout = 0;
    while (HAL_GPIO_ReadPin(GPIOA, ECHO_PIN)) {
        if (++timeout > 100000) return -1;
    }
    uint32_t duration = micros() - start;

    /* 声速 340m/s，除以 2 往返 */
    return duration * 0.034 / 2;
}

/* 舵机开盖 */
void Lid_Open(void) {
    __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_3, 120); /* 90度 */
}

/* 舵机关盖 */
void Lid_Close(void) {
    __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_3, 25);  /* 0度 */
}

/* 满桶百分比 */
float GetFillPercent(float distance_cm) {
    if (distance_cm >= 30) return 0;     /* 空桶 */
    if (distance_cm <= FULL_THRESHOLD) return 100; /* 满桶 */
    return (30 - distance_cm) / (30 - FULL_THRESHOLD) * 100;
}

int main(void) {
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_TIM3_Init();
    MX_I2C1_Init();
    OLED_Init();

    uint32_t last_pir_time = 0;
    uint8_t lid_open = 0;

    while (1) {
        /* 1. 检测红外人体感应 */
        if (HAL_GPIO_ReadPin(GPIOA, PIR_PIN) == GPIO_PIN_SET) {
            if (!lid_open) {
                Lid_Open();
                lid_open = 1;
                Ozone_On();  /* 开盖时启动除臭 */
                UV_On();     /* 启动杀菌 */
            }
            last_pir_time = HAL_GetTick();
        }

        /* 2. 人员离开 3 秒后关盖 */
        if (lid_open && (HAL_GetTick() - last_pir_time > LID_CLOSE_DELAY)) {
            Lid_Close();
            lid_open = 0;
            Ozone_Off();
            UV_Off();
        }

        /* 3. 超声波满桶检测 */
        float dist = HC_SR04_GetDistance();
        float fill = GetFillPercent(dist);
        OLED_ShowFillPercent((uint8_t)fill);

        /* 满桶报警 */
        if (fill >= 90) {
            Buzzer_Alert(); /* 蜂鸣器提示 */
        }

        delay(200);
    }
}
