核心代码 - PWM 调速与堵转保护

代码说明：TIM2 输出 10kHz PWM 驱动 AO3400 MOS。ADC 采样电流实现堵转保护。按键切换 3 挡模式。 
/* 手持电风扇 - PWM 调速主控程序 */

#include "fan_control.h"
#include "oled_display.h"
#include "battery.h"

/* 风扇挡位 */
typedef enum {
    FAN_OFF   = 0,
    FAN_LOW   = 30,    /* 微风 30% */
    FAN_MED   = 60,    /* 中档 60% */
    FAN_HIGH  = 100    /* 强风 100% */
} FanSpeed;

static FanSpeed current_speed = FAN_OFF;
static uint8_t speed_table[4] = {0, 30, 60, 100};
static uint8_t speed_index = 0;

/* 堵转电流阈值 */
#define STALL_CURRENT_THRESH  1500  /* ADC 值，约 500mA */

/* TIM2 初始化: 10kHz PWM */
void pwm_init(void) {
    RCC->APB1ENR |= RCC_APB1ENR_TIM2EN;
    TIM2->PSC = 0;                    /* 72MHz */
    TIM2->ARR = 7200 - 1;             /* 72MHz/7200 = 10kHz */
    TIM2->CCMR1 = TIM_CCMR1_OC1M_1 | TIM_CCMR1_OC1M_2;
    TIM2->CCER = TIM_CCER_CC1E;
    TIM2->BDTR = TIM_BDTR_MOE;
    TIM2->CCR1 = 0;
    TIM2->CR1 = TIM_CR1_CEN;
}

/* 设置风扇速度 */
void fan_set_speed(uint8_t percent) {
    uint32_t pulse = (uint32_t)7200 * percent / 100;
    TIM2->CCR1 = pulse;
    current_speed = (FanSpeed)percent;
}

/* 当前挡位文本 */
const char* fan_speed_text(void) {
    switch (current_speed) {
        case FAN_OFF:  return "关机";
        case FAN_LOW:  return "1档 微风";
        case FAN_MED:  return "2档 中档";
        case FAN_HIGH: return "3档 强风";
        default:       return "未知";
    }
}

/* 按键中断 - 模式切换 */
void EXTI0_IRQHandler(void) {
    if (EXTI->PR & EXTI_PR_PR0) {
        EXTI->PR = EXTI_PR_PR0;
        /* 切换挡位: OFF -> LOW -> MED -> HIGH -> OFF */
        speed_index = (speed_index + 1) % 4;
        fan_set_speed(speed_table[speed_index]);

        /* 更新 OLED */
        oled_show_speed(fan_speed_text(), speed_table[speed_index]);
    }
}

/* 按键中断 - 开关 */
void EXTI1_IRQHandler(void) {
    if (EXTI->PR & EXTI_PR_PR1) {
        EXTI->PR = EXTI_PR_PR1;

        if (current_speed == FAN_OFF) {
            /* 开机: 回到上次挡位 */
            if (speed_index == 0) speed_index = 1;
            fan_set_speed(speed_table[speed_index]);
        } else {
            /* 关机 */
            speed_index = 0;
            fan_set_speed(FAN_OFF);
        }

        oled_show_speed(fan_speed_text(), current_speed);
    }
}

/* 堵转检测 (周期 200ms) */
void stall_detection_task(void) {
    if (current_speed == FAN_OFF) return;

    uint32_t adc_val = 0;
    for (int i = 0; i < 8; i++) {
        ADC1->CR2 |= ADC_CR2_SWSTART;
        while (!(ADC1->SR & ADC_SR_EOC));
        adc_val += ADC1->DR;
    }
    adc_val /= 8;

    if (adc_val > STALL_CURRENT_THRESH) {
        /* 堵转保护: 立即关闭 */
        fan_set_speed(0);
        speed_index = 0;
        oled_show_warning("堵转保护!");

        /* 尝试 3 次恢复 */
        for (int retry = 0; retry < 3; retry++) {
            delay_ms(2000);
            fan_set_speed(speed_table[1]);  /* 低速启动 */
            delay_ms(500);
            /* 检查电流 */
            if (adc_read(ADC1, 1) < STALL_CURRENT_THRESH) {
                fan_set_speed(speed_table[speed_index ? speed_index : 1]);
                return;
            }
            fan_set_speed(0);
        }
        /* 3 次都失败，保持停机 */
        oled_show_warning("请检查风扇");
    }
}
