核心代码 - 滤芯寿命管理与自动冲洗

代码说明：根据累计水量计算各级滤芯消耗比例，结合使用天数估算滤芯寿命。每次制水结束后自动执行 RO 膜冲洗。 
/* 智能净水器 - 滤芯寿命管理与冲洗控制 */

#include "filter_manager.h"
#include "flow_meter.h"
#include "eeprom_sim.h"

/* 各级滤芯额定寿命 (升) */
#define PP_LIFETIME_L      2000.0f    /* PP 棉 2000L */
#define CARBON_LIFETIME_L  4000.0f    /* 活性炭 4000L */
#define RO_LIFETIME_L      6000.0f    /* RO 膜 6000L */

/* 冲洗参数 */
#define FLUSH_DURATION_MS  30000      /* 每次冲洗 30s */
#define DEEP_FLUSH_CYCLE   28800      /* 深度冲洗周期 8h (秒) */
#define DEEP_FLUSH_DURATION 60000     /* 深度冲洗 60s */

static FilterState filters[3];

/* 初始化滤芯状态 */
void filter_init(void) {
    filters[0].type = FILTER_PP;
    filters[0].lifetime_l = PP_LIFETIME_L;
    filters[0].used_l = eeprom_read_float(ADDR_PP_USED);

    filters[1].type = FILTER_CARBON;
    filters[1].lifetime_l = CARBON_LIFETIME_L;
    filters[1].used_l = eeprom_read_float(ADDR_CARBON_USED);

    filters[2].type = FILTER_RO;
    filters[2].lifetime_l = RO_LIFETIME_L;
    filters[2].used_l = eeprom_read_float(ADDR_RO_USED);
}

/* 更新滤芯寿命 (每制水 1L 调用) */
void filter_update_usage(float liters) {
    for (int i = 0; i < 3; i++) {
        filters[i].used_l += liters;
        float pct = (1.0f - filters[i].used_l / filters[i].lifetime_l) * 100.0f;
        if (pct < 0) pct = 0;
        filters[i].percent = pct;
    }

    /* 每 10L 保存一次 EEPROM */
    static float save_acc = 0;
    save_acc += liters;
    if (save_acc >= 10.0f) {
        save_acc = 0;
        eeprom_write_float(ADDR_PP_USED, filters[0].used_l);
        eeprom_write_float(ADDR_CARBON_USED, filters[1].used_l);
        eeprom_write_float(ADDR_RO_USED, filters[2].used_l);
    }
}

/* 检查是否需要更换 */
uint8_t filter_check_warning(void) {
    uint8_t warn = 0;
    for (int i = 0; i < 3; i++) {
        if (filters[i].percent < 10.0f) {
            warn |= (1 << i);
        }
    }
    return warn;
}

/* RO 膜自动冲洗 */
void ro_flush_execute(uint8_t deep) {
    uint32_t duration = deep ? DEEP_FLUSH_DURATION : FLUSH_DURATION_MS;

    VALVE_Close(INLET_VALVE);     /* 关闭进水阀 */
    PUMP_On();                    /* 启动增压泵 */
    VALVE_Open(FLUSH_VALVE);      /* 打开冲洗阀 */

    /* 冲洗计时 */
    uint32_t elapsed = 0;
    while (elapsed < duration) {
        OLED_ShowFlushStatus(elapsed, duration);
        delay_ms(100);
        elapsed += 100;

        /* 紧急停止检测 */
        if (Button_GetPress(KEY_FLUSH)) break;
    }

    VALVE_Close(FLUSH_VALVE);
    PUMP_Off();
    VALVE_Open(INLET_VALVE);
}

/* 主循环冲洗调度 */
void ro_flush_scheduler(void) {
    static uint32_t last_deep_flush = 0;
    uint32_t now = HAL_GetTick() / 1000;

    /* 每次制水结束后的标准冲洗在制水流程中调用 */
    /* 深度冲洗: 每 8 小时 */
    if (now - last_deep_flush >= DEEP_FLUSH_CYCLE) {
        last_deep_flush = now;
        ro_flush_execute(1);     /* 深度冲洗 */
    }
}
