软件架构 - FreeRTOS 任务分配

软件架构说明：采用 FreeRTOS 实时操作系统，按功能划分为三个独立任务， 利用消息队列实现任务间通信，确保系统实时性和稳定性。 
/* 智能家居主程序任务结构 (FreeRTOS) */

/* 任务 1：传感器采集（优先级 2，周期 2s） */
void vTask_SensorCollect(void *pvParam) {
    for (;;) {
        DHT11_ReadData();              // 读取温湿度
        BH1750_ReadLight();            // 读取光照
        MQTT_Publish_SensorData();     // 发布到 MQTT
        OLED_UpdateDisplay();          // 更新显示屏
        vTaskDelay(pdMS_TO_TICKS(2000));
    }
}

/* 任务 2：MQTT 通信（优先级 3，周期 100ms） */
void vTask_MQTT_Handle(void *pvParam) {
    for (;;) {
        ESP8266_ProcessBuffer();       // 处理 WiFi 数据
        if (mqtt_msg_available()) {
            handle_mqtt_command();     // 解析云端指令
        }
        vTaskDelay(pdMS_TO_TICKS(100));
    }
}

/* 任务 3：自动联动控制（优先级 1，周期 500ms） */
void vTask_AutoControl(void *pvParam) {
    for (;;) {
        if (lux < 50) Relay_On(LIGHT_RELAY);
        else           Relay_Off(LIGHT_RELAY);
        if (temp > 30) Relay_On(FAN_RELAY);
        else           Relay_Off(FAN_RELAY);
        vTaskDelay(pdMS_TO_TICKS(500));
    }
}
