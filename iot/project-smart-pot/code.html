核心代码 - 土壤湿度阈值与自动浇水

代码说明：ADC 采集土壤湿度百分比，低于阈值启动水泵， 高于停止。DHT11 温湿度辅助判断是否适合浇水（低温不浇）。 
/* 智能花盆 - 自动浇水逻辑 (HAL 库) */

#include <stdio.h>

/* 阈值定义 */
#define SOIL_DRY_THRESHOLD  30   /* 低于 30% 开始浇水 */
#define SOIL_WET_THRESHOLD  70   /* 高于 70% 停止浇水 */
#define PUMP_RUN_SEC        5    /* 每次最长浇水 5 秒 */
#define ADC_RESOLUTION      4096

/* 获取土壤湿度百分比 */
uint8_t Soil_GetMoisture(void) {
    HAL_ADC_Start(&hadc1);
    HAL_ADC_PollForConversion(&hadc1, 50);
    uint16_t adc = HAL_ADC_GetValue(&hadc1);
    HAL_ADC_Stop(&hadc1);
    /* YL-69: 干燥时 ADC 高，湿润时低，反转 */
    uint8_t pct = 100 - (adc * 100 / ADC_RESOLUTION);
    return pct;
}

/* 光照检测 (lux 估算) */
uint16_t Light_GetLux(void) {
    HAL_ADC_Start(&hadc2);
    HAL_ADC_PollForConversion(&hadc2, 50);
    uint16_t adc = HAL_ADC_GetValue(&hadc2);
    HAL_ADC_Stop(&hadc2);
    /* 粗略换算 */
    return (uint16_t)(adc * 1000 / ADC_RESOLUTION);
}

/* 水泵控制 */
void Pump_On(void) {
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_4, GPIO_PIN_RESET); /* MOS 低电平导通 */
}

void Pump_Off(void) {
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_4, GPIO_PIN_SET);
}

/* 自动浇水任务 (每 10 秒执行一次) */
void AutoWater_Task(void) {
    uint8_t moisture = Soil_GetMoisture();
    float temp, humi;

    DHT11_Read(&temp, &humi);

    /* 低温 (< 5°C) 不浇水防冻 */
    if (temp < 5.0f) {
        Pump_Off();
        return;
    }

    if (moisture < SOIL_DRY_THRESHOLD) {
        Pump_On();
        HAL_Delay(PUMP_RUN_SEC * 1000);
        /* 再次检测，若仍低于阈值继续 */
        uint8_t recheck = Soil_GetMoisture();
        if (recheck >= SOIL_WET_THRESHOLD) {
            Pump_Off();
        }
        /* 超时保护：最多运行 30 秒 */
        for (int i = 0; i < 6; i++) {
            if (Soil_GetMoisture() >= SOIL_WET_THRESHOLD) {
                Pump_Off();
                break;
            }
            HAL_Delay(5000);
        }
        Pump_Off();
    } else {
        Pump_Off();
    }
}

/* 蓝牙推送养护报告 */
void Send_Report(void) {
    char buf[128];
    uint8_t m = Soil_GetMoisture();
    uint16_t l = Light_GetLux();
    float t, h;
    DHT11_Read(&t, &h);

    snprintf(buf, sizeof(buf),
        "植物报告 | 土壤:%d%% | 光照:%dlux | %d°C %d%%RH\r\n",
        m, l, (int)t, (int)h);
    HC05_SendString(buf);
}
