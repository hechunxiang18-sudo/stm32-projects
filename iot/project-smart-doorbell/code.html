核心软件代码

代码说明：以下展示智能门铃的核心模块，包括 PIR 检测、OV7670 摄像头抓拍、ESP8266 WiFi 推送、音频采集与播放。 
/* ========== 智能门铃 - 主程序框架 ========== */

#include "ov7670.h"
#include "esp8266.h"
#include "sd_card.h"
#include "max9814.h"
#include "lm386.h"

#define PIR_PIN       GPIO_PIN_0
#define PIR_PORT      GPIOA
#define IR_LED_PIN    GPIO_PIN_15
#define IR_LED_PORT   GPIOA

volatile uint8_t g_pirTriggered = 0;
volatile uint8_t g_doorbellPressed = 0;
uint8_t g_imageBuffer[320 * 240];  /* 图像缓冲区（实际需外部SRAM） */

/* ===== 1. PIR 中断处理 ===== */
void EXTI0_IRQHandler(void) {
    if (EXTI_GetITStatus(EXTI_Line0) != RESET) {
        if (GPIO_ReadInputDataBit(PIR_PORT, PIR_PIN) == SET) {
            g_pirTriggered = 1;  /* 标记有人接近 */
        }
        EXTI_ClearITPendingBit(EXTI_Line0);
    }
}

/* ===== 2. OV7670 图像采集（FIFO模式） ===== */
uint8_t Camera_Capture(uint8_t *buffer) {
    /* 1. 复位FIFO写指针 */
    OV7670_FIFO_Reset();

    /* 2. 启动FIFO写入（允许OV7670写入FIFO） */
    OV7670_FIFO_WriteEnable(SET);
    HAL_Delay(100);  /* 等待一帧采集完成，实际需查Vsync中断 */

    /* 3. 停止写入 */
    OV7670_FIFO_WriteEnable(RESET);

    /* 4. 复位FIFO读指针 */
    OV7670_FIFO_ReadReset();

    /* 5. 读取FIFO数据（QVGA 320x240, 每个像素2字节RGB565） */
    uint32_t pixelCount = 320 * 240;
    uint16_t *ptr = (uint16_t *)buffer;

    for (uint32_t i = 0; i < pixelCount; i++) {
        /* 从FIFO读取一个像素的高8位和低8位 */
        uint8_t hi = OV7670_FIFO_ReadByte();
        uint8_t lo = OV7670_FIFO_ReadByte();
        *ptr++ = (hi << 8) | lo;
    }

    return 1;
}

/* ===== 3. JPEG 压缩（简化的 DCT 变换示意） ===== */
/* 实际使用 JPEG 库或串口发送给 ESP8266 处理 */

/* ===== 4. ESP8266 WiFi 图片推送 ===== */
void WiFi_PushImage(uint8_t *jpegData, uint32_t jpegLen) {
    char cmd[256];

    /* AT 指令模式：通过 TCP 发送图片到服务器 */
    ESP8266_SendCmd("AT+CIPSTART=\"TCP\",\"api.example.com\",8080\r\n");
    HAL_Delay(1000);

    /* 构建 HTTP POST 请求 */
    sprintf(cmd, "POST /upload HTTP/1.1\r\n");
    ESP8266_SendCmd(cmd);
    sprintf(cmd, "Host: api.example.com\r\n");
    ESP8266_SendCmd(cmd);
    sprintf(cmd, "Content-Length: %lu\r\n\r\n", jpegLen);
    ESP8266_SendCmd(cmd);

    /* 发送JPEG数据 */
    ESP8266_SendData(jpegData, jpegLen);
}

/* ===== 5. 音频采集与播放（PWM 方式） ===== */
void Audio_StartCapture(void) {
    /* 启动 TIM1 PWM，驱动 MAX9814 采样 */
    HAL_TIM_PWM_Start(&htim1, TIM_CHANNEL_1);
    /* 采样率 16kHz，8位PCM */
}

void Audio_Play(uint8_t *data, uint32_t len) {
    /* 通过 TIM3 PWM 播放音频数据 */
    for (uint32_t i = 0; i < len; i++) {
        TIM3->CCR1 = data[i];  /* 8位PWM占空比 */
        HAL_Delay_us(62);      /* 16kHz 采样周期 */
    }
}

/* ===== 6. 红外夜视控制 ===== */
void IR_NightVision_Control(void) {
    /* 读取光敏电阻（ADC）判断环境亮度 */
    HAL_ADC_Start(&hadc1);
    if (HAL_ADC_PollForConversion(&hadc1, 100) == HAL_OK) {
        uint16_t adcVal = HAL_ADC_GetValue(&hadc1);
        /* 低于阈值开启红外补光 */
        if (adcVal < 500) {
            GPIO_SetBits(IR_LED_PORT, IR_LED_PIN);
        } else {
            GPIO_ResetBits(IR_LED_PORT, IR_LED_PIN);
        }
    }
}

/* ===== 7. 门铃按键中断 ===== */
void EXTI15_10_IRQHandler(void) {
    /* 门铃按键处理 */
    if (EXTI_GetITStatus(EXTI_Line15) != RESET) {
        g_doorbellPressed = 1;
        EXTI_ClearITPendingBit(EXTI_Line15);
    }
}

/* ===== 8. 主循环 ===== */
int main(void) {
    HAL_Init();
    SystemClock_Config();

    EXTI0_Init();          /* PIR 中断 */
    EXTI15_Init();         /* 门铃按键中断 */
    UART2_Init(115200);    /* ESP8266 */
    SPI2_Init();           /* SD 卡 */
    TIM1_PWM_Init();       /* 音频采样 */
    TIM3_PWM_Init();       /* 音频播放 */

    OV7670_Init(OV7670_QVGA, OV7670_RGB565);
    ESP8266_Init();
    SDCard_Init();

    /* WiFi 连接 */
    ESP8266_ConnectWiFi("WiFi_SSID", "WiFi_PASSWORD");
    ESP8266_SendCmd("AT+CIPMUX=0\r\n");

    while (1) {
        /* 夜视控制 */
        IR_NightVision_Control();

        /* 1. PIR 触发或门铃按键 -> 抓拍并推送 */
        if (g_pirTriggered || g_doorbellPressed) {
            g_pirTriggered = 0;
            g_doorbellPressed = 0;

            /* 抓拍图片 */
            Camera_Capture(g_imageBuffer);

            /* 保存到SD卡 */
            SDCard_SaveImage(g_imageBuffer);

            /* 通过 WiFi 推送 */
            WiFi_PushImage(g_imageBuffer, 320 * 240 * 2);

            /* 触发音频提示 */
            Audio_Play("doorbell.wav");
        }

        /* 2. 定时刷新（每5秒检查一次） */
        HAL_Delay(5000);
    }
}
