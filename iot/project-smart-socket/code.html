核心代码 - 电能计量与过载保护

核心逻辑：HLW8032 通过串口主动上报电压、电流、功率和电量数据，STM32 解析后通过 ESP8266 上传到 MQTT 服务器。实时监测功率，超过阈值自动断开继电器。 
/* HLW8032 数据帧解析 */
typedef struct {
    uint32_t voltage;    /* 电压 x100 */
    uint32_t current;    /* 电流 x1000 */
    uint32_t power;      /* 功率 x100 */
    uint32_t energy;     /* 电量 x100 */
} HLW8032_Data;

HLW8032_Data hlw_data;

/* 解析 HLW8032 主动上报帧 (24 字节) */
uint8_t HLW8032_ParseFrame(uint8_t *buf) {
    if (buf[0] != 0x55 || buf[23] != Checksum_Calc(buf, 23))
        return 0;

    hlw_data.voltage = ((uint32_t)buf[1] << 8) | buf[2];
    hlw_data.current = ((uint32_t)buf[3] << 16) | ((uint32_t)buf[4] << 8) | buf[5];
    hlw_data.power   = ((uint32_t)buf[6] << 16) | ((uint32_t)buf[7] << 8) | buf[8];
    hlw_data.energy  = ((uint32_t)buf[9] << 16) | ((uint32_t)buf[10] << 8) | buf[11];
    return 1;
}

/* 过载保护检测 (每 200ms 调用) */
void Overload_Check(void) {
    uint32_t power_w = hlw_data.power / 100;  /* 转换为瓦特 */

    if (power_w > OVERLOAD_THRESHOLD) {        /* 如 2000W */
        Relay_Off();                            /* 立即断电 */
        overload_alarm = 1;                     /* 设置过载标志 */
        MQTT_Publish_Alarm("OVERLOAD", power_w); /* 发送告警 */
    }

    /* 自动恢复 */
    if (overload_alarm && power_w < OVERLOAD_CLEAR) {
        overload_alarm = 0;                     /* 清除过载标志 */
    }
}
