核心代码 - 水泵控制与水位保护

代码说明：主循环中读取水位值，依据水位和 PIR 触发状态控制水泵 PWM 输出和安全保护。 
/* 智能宠物饮水机 - 水泵控制与水位保护 */

#include "fountain_control.h"
#include "water_level.h"
#include "pir_sensor.h"
#include "oled_display.h"

/* 水泵工作模式 */
typedef enum {
    MODE_CONTINUOUS,    /* 连续循环 */
    MODE_INTERVAL,      /* 间歇运行 (开10min/停30min) */
    MODE_SMART,         /* 智能模式 (PIR 触发) */
    MODE_SLEEP          /* 睡眠待机 */
} FountainMode;

static FountainMode current_mode = MODE_SMART;
static uint8_t pump_running = 0;
static uint8_t water_ok = 0;

/* PWM 参数 */
#define PUMP_TIMER    TIM2
#define PUMP_CHANNEL  1
#define PWM_PERIOD    7199   /* 72MHz / (7199+1) = 10kHz */

/* 水泵初始化 */
void pump_init(void) {
    RCC->APB1ENR |= RCC_APB1ENR_TIM2EN;
    PUMP_TIMER->PSC = 0;
    PUMP_TIMER->ARR = PWM_PERIOD;
    PUMP_TIMER->CCMR1 = TIM_CCMR1_OC1M_1 | TIM_CCMR1_OC1M_2;  /* PWM 模式1 */
    PUMP_TIMER->CCER = TIM_CCER_CC1E;
    PUMP_TIMER->BDTR = TIM_BDTR_MOE;
    PUMP_TIMER->CR1 = TIM_CR1_CEN;
}

/* 设置水泵流量 0-100% */
void pump_set_speed(uint8_t percent) {
    uint32_t pulse = (uint32_t)PWM_PERIOD * percent / 100;
    PUMP_TIMER->CCR1 = pulse;
    pump_running = (percent > 0) ? 1 : 0;
}

/* 水位监测 */
uint8_t water_level_check(void) {
    uint32_t adc_val = 0;
    for (int i = 0; i < 8; i++) {
        adc_val += adc_read(ADC1, 1);
        delay_us(100);
    }
    adc_val /= 8;

    uint8_t percent = (uint8_t)(adc_val * 100 / 4095);
    water_ok = (percent > 15) ? 1 : 0;

    if (!water_ok) {
        pump_set_speed(0);     /* 立即停泵 */
        Buzzer_On();
        OLED_ShowWarning("缺水! 请加水");
    } else {
        Buzzer_Off();
    }

    return percent;
}

/* 智能模式主逻辑 */
void fountain_smart_loop(void) {
    water_level_check();

    if (!water_ok) {
        delay_ms(1000);
        return;
    }

    switch (current_mode) {
        case MODE_CONTINUOUS:
            pump_set_speed(70);
            break;

        case MODE_INTERVAL:
            /* 10 分钟运行，30 分钟停止 */
            static uint16_t interval_counter = 0;
            interval_counter++;
            if (interval_counter < 600) {       /* 10分钟 = 600秒 */
                pump_set_speed(70);
            } else {
                pump_set_speed(0);
                if (interval_counter >= 2400)   /* 30分钟 = 2400秒 */
                    interval_counter = 0;
            }
            break;

        case MODE_SMART:
            /* PIR 触发时开启，持续 5 分钟后自动减小流量 */
            if (pir_detected()) {
                pump_set_speed(80);
                uint16_t active_time = 0;
                while (active_time < 300) {      /* 持续 5 分钟 */
                    if (!water_level_check()) break;
                    delay_ms(1000);
                    active_time++;
                }
                pump_set_speed(30);              /* 转入低流量维持 */
            } else {
                pump_set_speed(30);              /* 无人时低流量循环 */
            }
            break;

        case MODE_SLEEP:
            pump_set_speed(0);
            break;
    }

    /* 更新 OLED 显示 */
    oled_show_status(current_mode, water_ok, pump_running);
}

/* UV 杀菌定时任务 (每 6 小时运行 15 分钟) */
void uv_schedule_task(void) {
    static uint32_t last_uv_time = 0;
    uint32_t now = HAL_GetTick() / 1000;

    if (now - last_uv_time >= 21600) {   /* 6 小时 */
        last_uv_time = now;
        GPIO_SetPin(GPIOA, GPIO_PIN_3);   /* 开启 UV */
        delay_ms(900000);                  /* 15 分钟 */
        GPIO_ResetPin(GPIOA, GPIO_PIN_3);  /* 关闭 UV */
    }
}
