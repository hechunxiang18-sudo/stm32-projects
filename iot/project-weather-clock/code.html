核心代码 - NTP 校时与天气 API 解析

代码说明：ESP8266 AT 指令连接 WiFi，通过 TCP 获取 NTP 时间 (SNTP) 和 天气 API JSON 数据。cJSON 库解析天气报文，更新 RTC 和 OLED 显示。 
/* WiFi 天气时钟 - NTP 校时与天气获取 (HAL 库) */

#include <stdio.h>
#include <string.h>
#include "cJSON.h"

/* NTP 服务器 */
#define NTP_SERVER   "ntp.aliyun.com"
#define NTP_PORT     123

/* 天气 API */
#define WEATHER_URL  "api.seniverse.com/v3/weather/now.json?key=YOUR_KEY&location=beijing"

/* NTP 时间戳获取 */
uint32_t Get_NTP_Time(void) {
    uint8_t ntp_pkt[48] = {0};
    ntp_pkt[0] = 0x1B;  /* NTP v3, client mode */

    /* ESP8266 TCP 连接 */
    ESP8266_TCPConnect(NTP_SERVER, NTP_PORT);

    /* 发送 NTP 请求 */
    ESP8266_SendData(ntp_pkt, 48);

    /* 接收响应 (超时 5s) */
    uint8_t buf[128];
    if (!ESP8266_ReceiveData(buf, 128, 5000)) return 0;

    /* 解析 NTP 时间戳 (字节 40-43) */
    uint32_t sec = (buf[40] << 24) | (buf[41] << 16)
                 | (buf[42] << 8)  | buf[43];

    /* NTP 到 Unix 时间偏移 2208988800 */
    sec -= 2208988800UL;

    /* 东八区 +8 */
    sec += 8 * 3600;

    return sec;
}

/* 设置 RTC 时间 */
void RTC_SetFromNTP(uint32_t unix_sec) {
    RTC_TimeTypeDef time;
    RTC_DateTypeDef date;

    /* 计算日期时间 (简化，使用本地函数) */
    UnixToDateTime(unix_sec, &date, &time);

    HAL_RTC_SetTime(&hrtc, &time, RTC_FORMAT_BIN);
    HAL_RTC_SetDate(&hrtc, &date, RTC_FORMAT_BIN);
}

/* 获取并解析天气 JSON */
uint8_t Fetch_Weather(float *temp, const char *weather, uint8_t *humidity) {
    char http_req[256];
    snprintf(http_req, sizeof(http_req),
        "GET /v3/weather/now.json?key=KEY&location=beijing HTTP/1.1\r\n"
        "Host: api.seniverse.com\r\n"
        "Connection: close\r\n\r\n");

    ESP8266_TCPConnect("api.seniverse.com", 80);
    ESP8266_SendString(http_req);

    char resp[1024];
    if (!ESP8266_ReceiveData((uint8_t*)resp, 1024, 10000)) return 0;

    /* 提取 JSON 部分 (跳过 HTTP 头) */
    char *json = strstr(resp, "\r\n\r\n");
    if (!json) return 0;
    json += 4;

    /* cJSON 解析 */
    cJSON *root = cJSON_Parse(json);
    if (!root) return 0;

    cJSON *results = cJSON_GetObjectItem(root, "results");
    cJSON *now = cJSON_GetObjectItem(cJSON_GetArrayItem(results, 0), "now");

    if (temp)     *temp     = atof(cJSON_GetObjectItem(now, "temperature")->valuestring);
    if (humidity) *humidity = atoi(cJSON_GetObjectItem(now, "humidity")->valuestring);
    if (weather)  strcpy(weather, cJSON_GetObjectItem(now, "text")->valuestring);

    cJSON_Delete(root);
    return 1;
}

/* 自动亮度调节 */
void Adjust_Brightness(void) {
    HAL_ADC_Start(&hadc1);
    HAL_ADC_PollForConversion(&hadc1, 50);
    uint16_t adc = HAL_ADC_GetValue(&hadc1);
    HAL_ADC_Stop(&hadc1);

    /* OLED 对比度 0-255 */
    uint8_t contrast = (uint8_t)(adc * 200 / 4096 + 20);
    OLED_SetContrast(contrast);
}

/* 闹钟检查 */
void Check_Alarm(RTC_HandleTypeDef *hrtc) {
    RTC_TimeTypeDef time;
    HAL_RTC_GetTime(hrtc, &time, RTC_FORMAT_BIN);

    /* 预设闹钟 07:00 */
    if (time.Hours == 7 && time.Minutes == 0 && time.Seconds == 0) {
        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_0, GPIO_PIN_SET);
        HAL_Delay(3000);
        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_0, GPIO_PIN_RESET);
    }
}
