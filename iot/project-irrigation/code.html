核心固件代码

/* 智能灌溉控制逻辑 */

#define DRY_THRESHOLD  30     // 土壤湿度低于 30% 开启灌溉
#define WET_THRESHOLD  70     // 湿度高于 70% 停止灌溉
#define MAX_IRRIGATE_SEC 600  // 单次最大灌溉 10 分钟

void Irrigation_Task(void) {
    /* 1. 读取天气 API 数据 */
    uint8_t will_rain = ESP8266_GetRainForecast();
    if (will_rain) {
        ESP8266_MQTT_Publish("irrigation/status", "Rained skipped");
        return;  // 预报有雨，跳过灌溉
    }

    /* 2. 检测当前是否下雨 */
    if (RainSensor_Read() < RAIN_THRESHOLD) {
        ESP8266_MQTT_Publish("irrigation/status", "Raining skipped");
        return;
    }

    /* 3. 逐分区检测灌溉 */
    for (int zone = 0; zone < 8; zone++) {
        uint16_t moisture = SoilMoisture_Read(zone);

        if (moisture < DRY_THRESHOLD) {
            /* 开启该区域电磁阀 */
            Valve_Open(zone);
            ESP8266_MQTT_Publish("irrigation/zone", zone, "ON");

            /* 持续灌溉，直到湿度达标或超时 */
            uint32_t start = HAL_GetTick();
            while ((HAL_GetTick() - start) < MAX_IRRIGATE_SEC * 1000) {
                delay_ms(30000);  // 每 30s 检查一次
                moisture = SoilMoisture_Read(zone);
                if (moisture >= WET_THRESHOLD) break;
            }

            Valve_Close(zone);
            ESP8266_MQTT_Publish("irrigation/zone", zone, "OFF");

            /* 记录用水量 */
            uint32_t water = FlowMeter_Read();
            Log_WaterUsage(zone, water);
        }
    }
}
