核心代码 - 调光调色与自动亮度

核心逻辑：PWM 驱动白光和暖光两路混合实现色温调节，光敏 ADC 采样环境光后 自动计算目标亮度，实现恒照度补偿。 
/* 色温 PWM 混合计算 */
void Light_SetColorTemp(uint16_t kelvin) {
    /* 3000K ~ 6500K 映射到 warm_ratio 0.0~1.0 */
    float warm_ratio = 1.0f - (float)(kelvin - 3000) / 3500;
    float cool_ratio = 1.0f - warm_ratio;

    uint16_t warm_pulse = (uint16_t)(current_brightness * warm_ratio * 999);
    uint16_t cool_pulse = (uint16_t)(current_brightness * cool_ratio * 999);

    TIM_SetCompare2(TIM2, warm_pulse);  /* PA1 - 暖光 */
    TIM_SetCompare3(TIM2, cool_pulse);  /* PA2 - 冷光 */
}

/* 自动调光：根据环境光补偿 */
void AutoBrightness_Adjust(void) {
    uint16_t adc_val = ADC_GetValue(ADC1, ADC_Channel_0);

    /* 环境光越暗，台灯越亮（反比例映射）*/
    uint16_t target_brightness;
    if (adc_val < 200) {
        target_brightness = 80;   /* 环境很暗 → 80% */
    } else if (adc_val < 800) {
        target_brightness = 60;   /* 较暗 → 60% */
    } else if (adc_val < 2000) {
        target_brightness = 40;   /* 适中 → 40% */
    } else {
        target_brightness = 20;   /* 明亮 → 20% */
    }

    /* 渐变过渡到目标亮度 */
    if (current_brightness < target_brightness)
        current_brightness++;
    else if (current_brightness > target_brightness)
        current_brightness--;

    Light_SetBrightness(current_brightness);
}

/* 45 分钟休息提醒 (基于 SysTick 计时) */
uint32_t work_seconds = 0;

void RestReminder_Check(void) {
    work_seconds++;
    if (work_seconds >= 45 * 60) {      /* 45 分钟 */
        OLED_ShowString(0, 4, "Take a rest!");
        Buzzer_Twice();                  /* 蜂鸣提示 */
        work_seconds = 0;
    }
}
