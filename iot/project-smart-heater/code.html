核心代码 - PID 恒温控制

核心逻辑：采用位置式 PID 控制算法，根据 DS18B20 实测温度与目标温度的误差 计算控制量，通过继电器占空比调节加热功率，实现 ±0.5°C 的恒温精度。 
/* PID 结构体 */
typedef struct {
    float target_temp;    /* 目标温度 */
    float kp, ki, kd;     /* PID 参数 */
    float integral;       /* 积分累加 */
    float last_error;     /* 上次误差 */
    float output;         /* 输出 (0.0 ~ 1.0) */
} PID_Controller;

PID_Controller pid = {
    .target_temp = 50.0f,
    .kp = 15.0f, .ki = 0.2f, .kd = 5.0f,
    .integral = 0, .last_error = 0, .output = 0
};

/* PID 计算 (每 1 秒调用) */
void PID_Compute(PID_Controller *pid, float current_temp) {
    float error = pid->target_temp - current_temp;

    /* 积分分离：大误差时不积分 */
    if (fabs(error) < 10.0f) {
        pid->integral += error * 1.0f;  /* dt = 1s */
        if (pid->integral > 100)  pid->integral = 100;
        if (pid->integral < -100) pid->integral = -100;
    }

    float derivative = error - pid->last_error;
    pid->output = pid->kp * error + pid->ki * pid->integral + pid->kd * derivative;
    pid->last_error = error;

    /* 输出限幅 */
    if (pid->output > 1.0f) pid->output = 1.0f;
    if (pid->output < 0.0f) pid->output = 0.0f;

    /* 控制继电器: output > 0.5 加热，output < -0.1 停止 */
    if (pid->output > 0.15f)
        Relay_On();       /* 开始加热 */
    else
        Relay_Off();      /* 停止加热 */
}

/* 防干烧保护检测 */
void DryFire_Protect(void) {
    uint16_t water_level = ADC_GetValue(ADC1, ADC_Channel_0);
    if (water_level < DRY_FIRE_THRESHOLD) {
        Relay_Off();
        Buzzer_On();
        MQTT_Publish_Alarm("DRY_FIRE");
    }
}
