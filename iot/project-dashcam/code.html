核心代码 - 碰撞检测与片段锁定

代码说明：主循环中周期性读取 MPU6050 加速度数据，计算矢量加速度变化量，超过阈值则判定为碰撞事件，立即锁定当前录制文件。 
/* 智能行车记录仪 - 碰撞检测与文件锁定 */

#include "mpu6050.h"
#include "sd_card.h"
#include "dashcam.h"

#define G_SENSOR_THRESHOLD  2.5f    /* 碰撞加速度阈值 (g) */
#define LOCK_FILE_PREFIX    "EVENT_" /* 锁定文件前缀 */

/* 读取 MPU6050 并计算加速度矢量幅值 */
float get_accel_magnitude(void) {
    int16_t ax, ay, az;
    MPU6050_ReadAccel(&ax, &ay, &az);

    /* 转换为 g 单位 (±2g 量程: 16384 LSB/g) */
    float fx = ax / 16384.0f;
    float fy = ay / 16384.0f;
    float fz = az / 16384.0f;

    return sqrtf(fx*fx + fy*fy + fz*fz);
}

/* 碰撞检测与锁定任务 (周期 50ms) */
void vTask_CrashDetect(void *param) {
    float baseline = 1.0f;     /* 静止基准 (1g = 重力) */
    uint8_t trigger_count = 0;

    for (;;) {
        float mag = get_accel_magnitude();
        float delta = fabsf(mag - baseline);

        if (delta > G_SENSOR_THRESHOLD) {
            trigger_count++;
            if (trigger_count >= 3) {
                /* 连续 3 次超过阈值 → 确认碰撞 */
                dashcam_lock_current_file(LOCK_FILE_PREFIX);
                LED_Blink(LED_RED, 5);    /* LED 闪烁提示 */
                trigger_count = 0;
            }
        } else {
            if (trigger_count > 0) trigger_count--;
        }

        /* 动态更新基准值 (低通滤波) */
        baseline = baseline * 0.98f + mag * 0.02f;

        vTaskDelay(pdMS_TO_TICKS(50));
    }
}

/* 锁定当前正在写入的文件 */
void dashcam_lock_current_file(const char* prefix) {
    char old_name[32], new_name[32];

    sd_card_get_current_filename(old_name, sizeof(old_name));

    /* 生成锁定文件名: EVENT_001.AVI */
    static uint16_t event_num = 0;
    snprintf(new_name, sizeof(new_name), "%s%03d.AVI",
             prefix, ++event_num);

    sd_card_rename(old_name, new_name);
    sd_card_set_readonly(new_name);

    /* 保存事件日志到 CSV */
    dashcam_log_event(event_num, get_rtc_timestamp());
}
