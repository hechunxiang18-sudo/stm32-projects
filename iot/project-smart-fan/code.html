核心代码 - BLDC 六步换向与温控调速

核心逻辑：通过霍尔传感器读取转子位置，按六步换向表依次导通不同相位的 MOS 管， 配合 PWM 占空比调节实现转速控制。NTC 温度检测自动修正目标转速。 
/* BLDC 六步换向表 (霍尔编码 -> 导通相) */
const uint8_t bldc_comm_table[8] = {
    0,      /* 非法态 */
    0x0C,   /* 001: U+V- */
    0x14,   /* 010: U-W+ */
    0x18,   /* 011: V-U+ */
    0x06,   /* 100: V-W- */
    0x09,   /* 101: W-U- */
    0x22,   /* 110: W+V- */
    0       /* 非法态 */
};

/* 设置换相 */
void BLDC_Commutation(uint8_t hall_val) {
    if (hall_val == 0 || hall_val == 7) return;  /* 非法 */

    uint8_t phase = bldc_comm_table[hall_val];
    /* 根据相位表设置 GPIO / PWM 输出 */
    TIM_SetCompare1(TIM2, (phase & 0x01) ? speed_pulse : 0);
    TIM_SetCompare2(TIM2, (phase & 0x02) ? speed_pulse : 0);
    TIM_SetCompare3(TIM2, (phase & 0x04) ? speed_pulse : 0);
    TIM_SetCompare4(TIM2, (phase & 0x08) ? speed_pulse : 0);
    TIM_SetCompare1(TIM3, (phase & 0x10) ? speed_pulse : 0);
    TIM_SetCompare2(TIM3, (phase & 0x20) ? speed_pulse : 0);
}

/* 霍尔中断处理 (转子位置变化触发) */
void EXTI0_IRQHandler(void) {
    if (EXTI_GetITStatus(EXTI_Line0)) {
        uint8_t hall = GPIO_ReadInputDataBits(GPIOB,
            GPIO_Pin_0 | GPIO_Pin_1 | GPIO_Pin_2);
        BLDC_Commutation(hall);
        EXTI_ClearITPendingBit(EXTI_Line0);
    }
}

/* 温控自动调速 (每 5 秒) */
void Temp_Auto_Speed(void) {
    uint16_t adc = ADC_GetValue(ADC1, ADC_Channel_7);
    float temp = 25.0f + (3.3f * adc / 4095 - 0.5f) * 100.0f;

    if (temp < 24.0f)       target_speed = 0;   /* 低于 24°C 停止 */
    else if (temp < 26.0f)  target_speed = 2;   /* 26°C 低档 */
    else if (temp < 28.0f)  target_speed = 4;
    else if (temp < 30.0f)  target_speed = 6;
    else if (temp < 32.0f)  target_speed = 8;
    else                    target_speed = 12;  /* >32°C 最高档 */
}
