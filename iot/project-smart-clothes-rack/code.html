核心代码 - 雨感光感升降控制

核心逻辑：主循环中定时采样雨滴传感器和光敏电阻，根据阈值自动控制电机升降。 限位开关中断触发时立即停止电机，蓝牙和红外遥控可覆盖自动模式。 
/* 晾衣架状态 */
typedef enum { STOP, EXTENDING, RETRACTING } Rack_State;
Rack_State rack_state = STOP;
uint8_t rack_extended = 0;   /* 是否已伸出 */

/* 自动升降控制 (每 500ms 调用) */
void AutoRack_Control(void) {
    uint16_t rain_val = ADC_GetValue(ADC1, ADC_Channel_0);  /* 雨滴 */
    uint16_t light_val = ADC_GetValue(ADC1, ADC_Channel_1); /* 光照 */

    /* 下雨检测：雨滴值低于阈值表示有雨 */
    uint8_t is_raining = (rain_val < RAIN_THRESHOLD);

    /* 光照检测：高于阈值表示天亮 */
    uint8_t is_bright = (light_val > LIGHT_THRESHOLD);

    if (is_raining && rack_extended) {
        /* 下雨 & 已伸出 → 收回 */
        Rack_Retract();
    }
    else if (!is_raining && is_bright && !rack_extended) {
        /* 无雨 & 天亮 & 未伸出 → 伸出 */
        Rack_Extend();
    }
}

/* 电机控制 */
void Rack_Extend(void) {
    if (rack_state != STOP) return;
    if (GPIO_ReadInputDataBit(GPIOB, GPIO_Pin_0) == 0) {
        /* 上限位已触发，不能再伸 */
        return;
    }
    rack_state = EXTENDING;
    GPIO_SetBits(GPIOA, GPIO_Pin_5);   /* 方向：伸出 */
    TIM_SetCompare1(TIM3, 800);         /* PWM 80% 速度 */
}

void Rack_Retract(void) {
    if (rack_state != STOP) return;
    if (GPIO_ReadInputDataBit(GPIOB, GPIO_Pin_1) == 0) {
        /* 下限位已触发，不能再收 */
        return;
    }
    rack_state = RETRACTING;
    GPIO_ResetBits(GPIOA, GPIO_Pin_5); /* 方向：收回 */
    TIM_SetCompare1(TIM3, 800);
}

/* 限位开关中断 */
void EXTI0_IRQHandler(void) {  /* 上限位 */
    if (EXTI_GetITStatus(EXTI_Line0)) {
        TIM_SetCompare1(TIM3, 0);       /* 停止电机 */
        rack_state = STOP;
        rack_extended = 1;              /* 标记已伸出 */
        EXTI_ClearITPendingBit(EXTI_Line0);
    }
}
