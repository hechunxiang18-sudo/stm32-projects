/* 智能鱼缸 PID 恒温控制 */

static PID_t temp_pid = {
    .kp = 15.0f,
    .ki = 0.5f,
    .kd = 2.0f,
};
static float target_temp = 26.0f;  // 目标温度 26°C

void Aquarium_TempControl(void) {
    /* 读取水温（双传感器平均） */
    float t1 = DS18B20_Read(0);
    float t2 = DS18B20_Read(1);
    float current = (t1 + t2) / 2.0f;

    /* PID 计算 */
    float error = target_temp - current;
    temp_pid.integral += error * 0.5f;  // 每 0.5s 执行一次
    temp_pid.integral = CLAMP(temp_pid.integral, -50, 50);

    float derivative = (error - temp_pid.prev_error) / 0.5f;
    temp_pid.prev_error = error;

    float output = temp_pid.kp * error
                 + temp_pid.ki * temp_pid.integral
                 + temp_pid.kd * derivative;

    /* 输出控制（死区 ±0.3°C）*/
    if (fabs(error) < 0.3f) {
        Relay_Off(HEATER_RELAY);
        Relay_Off(COOLER_RELAY);
        return;
    }

    if (output > 0) {
        /* 需要加热 - PWM 控制加热棒功率 */
        uint16_t duty = (uint16_t)(output * 100);
        TIM_SetCompare(TIM2, CLAMP(duty, 0, 100));
        Relay_Off(COOLER_RELAY);
    } else {
        /* 需要制冷 - 开启半导体制冷 */
        Relay_On(COOLER_RELAY);
        Relay_Off(HEATER_RELAY);
    }
}

/* 灯光日出日落模拟 */
void Aquarium_LightCycle(void) {
    RTC_TimeTypeDef now;
    HAL_RTC_GetTime(&hrtc, &now, RTC_FORMAT_BIN);

    uint8_t h = now.Hours, m = now.Minutes;
    uint8_t minutes = h * 60 + m;

    /* 6:00 日出渐变 → 12:00 全亮 → 18:00 日落 → 22:00 关闭 */
    if (minutes >= 360 && minutes < 720) {
        uint16_t t = (minutes - 360) * 255 / 360;  // 线性渐变
        LED_SetBrightness(t);
    } else if (minutes >= 1080 && minutes < 1320) {
        uint16_t t = 255 - (minutes - 1080) * 255 / 240;
        LED_SetBrightness(t);
    } else if (minutes >= 1320 || minutes < 360) {
        LED_SetBrightness(0);  // 夜晚关灯
    } else {
        LED_SetBrightness(255);  // 白天全亮
    }
}
