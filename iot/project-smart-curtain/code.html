核心代码 - 步进电机控制

核心逻辑：28BYJ-48 步进电机采用四相八拍驱动方式（A-AB-B-BC-C-CD-D-DA）， 通过 ULN2003 驱动。光敏 ADC 定时采样，联动自动控制窗帘开合。 
/* 步进电机四相八拍驱动表 */
const uint8_t step_seq[8] = {
    0x08, 0x0C, 0x04, 0x06,  /* A, AB, B, BC */
    0x02, 0x03, 0x01, 0x09   /* C, CD, D, DA */
};

/* 左电机 GPIO (PB0-PB3) */
#define MOTOR_L_PORT  GPIOB
#define MOTOR_L_PINS  (GPIO_Pin_0|GPIO_Pin_1|GPIO_Pin_2|GPIO_Pin_3)

/* 右电机 GPIO (PB4-PB7) */
#define MOTOR_R_PORT  GPIOB
#define MOTOR_R_PINS  (GPIO_Pin_4|GPIO_Pin_5|GPIO_Pin_6|GPIO_Pin_7)

/* 步进电机走一步 */
void Stepper_Step(GPIO_TypeDef* port, uint16_t base_pin, uint8_t step_val) {
    uint16_t pins[] = {base_pin, base_pin << 1, base_pin << 2, base_pin << 3};
    for (int i = 0; i < 4; i++) {
        if (step_val & (1 << i))
            GPIO_SetBits(port, pins[i]);
        else
            GPIO_ResetBits(port, pins[i]);
    }
}

/* 光敏 ADC 读取与自动控制 */
void AutoCurtain_Control(void) {
    uint16_t adc_val = ADC_GetValue(ADC1, ADC_Channel_0);
    uint32_t lux = (uint32_t)adc_val * 100 / 4095;  /* 转换到近似 lux */

    static uint8_t curtain_open = 1;

    if (lux < 20 && curtain_open) {     /* 光线暗 → 关窗帘 */
        Stepper_Move(MOTOR_L_PORT, 512, STEP_CLOSE);
        Stepper_Move(MOTOR_R_PORT, 512, STEP_CLOSE);
        curtain_open = 0;
    }
    else if (lux > 60 && !curtain_open) { /* 光线亮 → 开窗帘 */
        Stepper_Move(MOTOR_L_PORT, 512, STEP_OPEN);
        Stepper_Move(MOTOR_R_PORT, 512, STEP_OPEN);
        curtain_open = 1;
    }
}
